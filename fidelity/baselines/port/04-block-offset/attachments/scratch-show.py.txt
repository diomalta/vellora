import json, sys

v = json.load(open(sys.argv[1]))
c = json.load(open(sys.argv[2]))['probes']
for p in v:
    print(p)
    for r in v[p]['runs']:
        ch = c[p].get(r['owner_id'] or '', {})
        cb = ch.get('baseline')
        d = (r['final_y'] - cb) if cb is not None else None
        print('   %-24s owner=%-9s L0=%8.3f final=%8.3f chrome=%s  final-chrome=%s' % (
            r['text'][:24], r['owner_id'], r['l0_y'], r['final_y'], cb,
            None if d is None else round(d, 3)))
    vw = {}
    for w in v[p]['vellora_pdf_words']:
        vw.setdefault(w['text'], w)
    for w in (v[p]['chrome_pdf_words'] or []):
        vv = vw.get(w['text'])
        if vv is None:
            continue
        print('      pdf %-12s chrome %.3f vellora %.3f  dy(pt) %+.3f' % (
            w['text'][:12], w['baseline'], vv['baseline'], vv['baseline'] - w['baseline']))
