"""REF-W distribution (tol 2 consensus) old vs new per fixture: median/mean |dy|, p50/p95/max."""
import json, sys, glob, os

old_run, new_run = sys.argv[1:3]


def stats(run):
    out = {}
    for f in sorted(glob.glob(run + '/oracles/*/*/plain/result.json')):
        name = f.split('/')[-3]
        d = json.load(open(f))
        lay = d['results'].get('layout') or {}
        for t in lay.get('tolerances', []):
            if t['tol'] == 2:
                se = t['subjectError']
                out[name] = (t['consensus'], se['medianDy'], se['meanAbsDy'], se['p50Distance'], se['p95Distance'], se['maxDistance'], se['medianDx'])
    return out


o, n = stats(old_run), stats(new_run)
print(f"{'fixture':26s} {'words':>9s} {'medianDy':>15s} {'mean|dy|':>13s} {'p50':>13s} {'p95':>13s} {'max':>15s}  (pt, tol-2 consensus)")
for k in sorted(n):
    a, b = o.get(k), n[k]
    if not a or None in a or None in b:
        continue
    print(f"{k:26s} {a[0]:4d}/{b[0]:4d} {a[1]:+7.2f}/{b[1]:+7.2f} {a[2]:6.2f}/{b[2]:6.2f} {a[3]:6.2f}/{b[3]:6.2f} {a[4]:6.2f}/{b[4]:6.2f} {a[5]:7.2f}/{b[5]:7.2f}")
