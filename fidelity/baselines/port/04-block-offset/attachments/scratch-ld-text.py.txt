"""Print layout-diff text events (per-layer dy) whose text matches, plus their owner box deltas."""
import json, sys

d = json.load(open(sys.argv[1]))
needles = sys.argv[2:]
bx = {b['path']: b for b in d['boxes']}
layers = [l['name'] for l in d['layers']] + ['P']
for ev in d['text_events']:
    if not any(n in ev['text'] for n in needles):
        continue
    pl = ev['per_layer']
    print(repr(ev['text'][:40]), ev['owner'][-70:], 'chrome', ev['chrome'])
    print('   text dy:', {k: (pl.get(k) or {}).get('dy') for k in layers})
    b = bx.get(ev['owner'])
    if b:
        print('   box  dy/dh:', {k: ((b['layers'].get(k) or {}).get('dy'), (b['layers'].get(k) or {}).get('dh')) for k in layers}, 'chrome', b['chrome'])
