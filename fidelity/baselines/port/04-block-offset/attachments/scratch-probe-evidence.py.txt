"""Per-item evidence for gating pdf-probe words (scratch, not part of the repo).

For every gating `pdf-probe|<fixture>|<word>#<k>|<axis>` item of a gate report, find the pdf-probe
pair (same k-th-occurrence rule as fidelity/gate/snapshot.mjs compactPdfProbe) in the old and the
new run, then read each word's BASELINE with PyMuPDF from the same two PDFs pdf-probe compared
(layout-diff/<fixture>/run-all/{vellora,chrome}.pdf): the span origin y, which does not depend on the
FontDescriptor /Ascent the producer wrote. pdf-probe's y is poppler `pdftotext -bbox` yMin =
baseline - Ascent x size; krilla writes OS/2 sTypoAscender (Liberation Sans 0.728 em, Serif 0.677 /
0.693 em) and Chrome hhea ascender (0.905 / 0.891 em), so pdf-probe dy carries +dA x size.

  fidelity/.venv/bin/python -I probe_evidence.py <gate-report.json> <old run> <new run> > evidence.json
"""
import json
import sys
import unicodedata

import pymupdf

gate_file, old_run, new_run = sys.argv[1:4]
gate = json.load(open(gate_file))

_cache = {}


def probe(run, fx):
    key = ('probe', run, fx)
    if key not in _cache:
        _cache[key] = json.load(open(f'{run}/layout-diff/{fx}/run-all/pdf-probe.json'))
    return _cache[key]


def words(run, fx, which):
    key = ('words', run, fx, which)
    if key not in _cache:
        doc = pymupdf.open(f'{run}/layout-diff/{fx}/run-all/{which}.pdf')
        out = []
        for pno, page in enumerate(doc):
            for w in page.get_text('rawdict')['blocks']:
                for line in w.get('lines', []):
                    for span in line['spans']:
                        # split span into words with their own origin x
                        chars = span['chars']
                        cur = []
                        for ch in chars + [None]:
                            if ch is None or ch['c'].isspace():
                                if cur:
                                    out.append({
                                        'page': pno + 1,
                                        'text': ''.join(c['c'] for c in cur),
                                        'x': cur[0]['origin'][0],
                                        'baseline': cur[0]['origin'][1],
                                        'size': span['size'],
                                        'font': span['font'],
                                        'ytop': cur[0]['bbox'][1],
                                    })
                                cur = []
                            else:
                                cur.append(ch)
        _cache[key] = out
    return _cache[key]


def pair_for(run, fx, k):
    text, _, occ = k.rpartition('#')
    occ = int(occ)
    n = 0
    for p in probe(run, fx)['pairs']:
        if unicodedata.normalize('NFC', p['text']) == text:
            n += 1
            if n == occ:
                return p
    return None


def baseline_of(run, fx, which, side):
    """PyMuPDF word with the same text on the same page whose x is nearest pdf-probe's xMin."""
    best = None
    for w in words(run, fx, which):
        if w['page'] != side['page'] or w['text'] != side['text']:
            continue
        d = abs(w['x'] - side['x']) + abs((w['ytop']) - side['y']) * 0.25
        if best is None or d < best[0]:
            best = (d, w)
    return None if best is None or best[0] > 6 else best[1]


def r3(v):
    return None if v is None else round(v, 3)


out = {}
for it in gate['items']:
    if not it.get('gating') or not it['key'].startswith('pdf-probe|'):
        continue
    _, fx, k, axis = it['key'].split('|')
    rec = {'key': it['key'], 'fixture': fx, 'word': k, 'axis': axis, 'probe_before': it['before'],
           'probe_after': it['after']}
    for label, run in (('old', old_run), ('new', new_run)):
        p = pair_for(run, fx, k)
        if p is None:
            rec[label] = None
            continue
        v = baseline_of(run, fx, 'vellora', {**p['vellora'], 'text': p['text']})
        c = baseline_of(run, fx, 'chrome', {**p['chrome'], 'text': p['text']})
        rec[label] = {
            'vellora_probe': p['vellora'], 'chrome_probe': p['chrome'],
            'probe_dx': p['dx_pt'], 'probe_dy': p['dy_pt'],
            'vellora_baseline': r3(v and v['baseline']), 'chrome_baseline': r3(c and c['baseline']),
            'vellora_x': r3(v and v['x']), 'chrome_x': r3(c and c['x']),
            'font': v and v['font'], 'size': r3(v and v['size']), 'chrome_font': c and c['font'],
            'baseline_dy': r3(v['baseline'] - c['baseline']) if v and c else None,
            'origin_dx': r3(v['x'] - c['x']) if v and c else None,
        }
    o, n = rec.get('old'), rec.get('new')
    if o and n and o.get('baseline_dy') is not None and n.get('baseline_dy') is not None:
        rec['abs_baseline_dy_change'] = r3(abs(n['baseline_dy']) - abs(o['baseline_dy']))
        rec['abs_origin_dx_change'] = r3(abs(n['origin_dx']) - abs(o['origin_dx']))
        rec['ascent_artefact_old'] = r3(o['probe_dy'] - o['baseline_dy'])
        rec['ascent_artefact_new'] = r3(n['probe_dy'] - n['baseline_dy'])
    out[it['key']] = rec
print(json.dumps(out, indent=1, ensure_ascii=False))
