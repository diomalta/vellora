"""Per fixture: does any vellora layer after raw Blitz move text y? (layout-diff diff.json)

For every text event with a B0 (raw Blitz placement of html_normalize's output) and a final-layer
value, count how many have final dy == B0 dy (|diff| <= 0.01 px), and N0 == B0, and P == final.
"""
import json, sys, glob

for f in sorted(glob.glob(sys.argv[1] + '/layout-diff/*/run-all/diff.json')):
    d = json.load(open(f))
    name = f.split('/')[-3]
    final = d['final_layer']
    tot = same_b0 = same_n0 = same_p = with_b0 = with_n0 = with_p = 0
    moved = []
    for ev in d['text_events']:
        pl = ev['per_layer']
        fin = (pl.get(final) or {}).get('dy')
        b0 = (pl.get('B0') or {}).get('dy')
        n0 = (pl.get('N0') or {}).get('dy')
        p = (pl.get('P') or {}).get('dy')
        if fin is None:
            continue
        tot += 1
        if b0 is not None:
            with_b0 += 1
            if abs(fin - b0) <= 0.01:
                same_b0 += 1
            else:
                moved.append((ev['text'][:30], round(b0, 3), round(fin, 3)))
        if n0 is not None and b0 is not None:
            with_n0 += 1
            if abs(n0 - b0) <= 0.01:
                same_n0 += 1
        if p is not None:
            with_p += 1
            if abs(p - fin) <= 0.01:
                same_p += 1
    print(f'{name:24s} runs {tot:4d}  final==B0 {same_b0}/{with_b0}  N0==B0 {same_n0}/{with_n0}  P==final {same_p}/{with_p}  moved-by-vellora {moved[:3]}')
