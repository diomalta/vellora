import json, glob, collections, sys

for f in sorted(glob.glob(sys.argv[1] + '/layout-diff/*/run-all/diff.json')):
    d = json.load(open(f))
    fin = d['final_layer']
    c = collections.Counter()
    for ev in d['text_events']:
        pl = ev['per_layer']
        a = (pl.get(fin) or {}).get('dy')
        b = (pl.get('B0') or {}).get('dy')
        if a is None or b is None:
            continue
        c[round(a - b, 2)] += 1
    print(f.split('/')[-3], dict(c))
