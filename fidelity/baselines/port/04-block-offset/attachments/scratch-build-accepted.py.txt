"""Scratch (not part of the repo): build accepted.json + accepted-evidence.json for port step 04.

  fidelity/.venv/bin/python -I build_accepted.py <gate-report.json (no acceptances)> <old run> <new run> \
      <out accepted.json> <out accepted-evidence.json>

Every gating item gets exactly one label (or the script lists it as unlabelled; absolute checks
such as code checks, canaries and layout-diff checks are never accepted).

Labels that FIT the merge rule's literal wording (GATE-A1: the old value was an oracle artefact):
  pdf-probe-ascent-artefact, pdf-probe-pairing-artefact, wpt-subpixel-raster-artefact.
Labels that DO NOT fit it (two errors cancelling; the orchestrator decides, as for steps 01/03):
  compensating-drift, misplaced-word-uniform-shift. The 5 pt split between these two is a triage aid
  (words that looked right before vs words already visibly off), not a rule claim.

  pdf-probe-ascent-artefact     (oracle artefact) a pdf-probe dy item whose word BASELINE did not get
                                worse beyond the 1 pt tolerance. pdf-probe's y is poppler `pdftotext
                                -bbox` yMin = baseline - FontDescriptor /Ascent x size; krilla writes
                                OS/2 sTypoAscender (Liberation Sans 0.728 em, Serif 0.677/0.693 em),
                                Chrome the hhea ascender (0.905/0.891 em), so pdf-probe dy = baseline
                                dy + dA x size. Baselines are PyMuPDF span origins of the SAME two PDFs
                                (independent of /Ascent).
  compensating-drift            the word/anchor/metric is worse because a removed vellora correction
                                (the constant 4 px = 3 pt PDF raise; the 2 px painted-row raise) or the
                                dropped block_offset had been cancelling raw Blitz box drift. The new
                                PDF baseline is the paginated layout baseline (pdf.rs adds nothing) and, in this
                                run, no vellora layer moves text y (layout-diff: final == B0 for the
                                fixture's runs); the residual is raw Blitz's own box position.
                                DOES NOT fit the merge rule's literal "false pass / oracle artefact".
  misplaced-word-uniform-shift  a pdf-probe word whose old baseline was already >= 5 pt (1.76 mm) from
                                Chrome (e.g. text-stress rows displaced by vellora's 8-vs-7 page breaks)
                                and that moved by exactly the removed 3 pt raise (+ block_offset).
  pdf-probe-pairing-artefact    the k-th-occurrence pair is not the same word instance (paired Chrome
                                word > 20 pt away horizontally in both runs).
  wpt-subpixel-raster-artefact  a WPT clean pass lost to a sub-pixel anti-aliasing difference (1 px)
                                in a reftest Chrome's own run of the same harness also fails.
"""
import html
import json
import sys
import unicodedata

import pymupdf

gate_file, old_run, new_run, out_accept, out_evidence = sys.argv[1:6]
gate = json.load(open(gate_file))
TOL = 1.0
MISPLACED_PT = 5.0  # an old baseline this far from Chrome (1.76 mm) was misplaced before the step
PAIR_DX_PT = 20.0  # a pair this far apart horizontally (both runs) is not the same word instance
_cache = {}


def probe(run, fx):
    key = ('probe', run, fx)
    if key not in _cache:
        _cache[key] = json.load(open(f'{run}/layout-diff/{fx}/run-all/pdf-probe.json'))
    return _cache[key]


def words(run, fx, which):
    key = ('words', run, fx, which)
    if key not in _cache:
        doc = pymupdf.open(f'{run}/layout-diff/{fx}/run-all/{which}.pdf')
        out = []
        for pno, page in enumerate(doc):
            for block in page.get_text('rawdict')['blocks']:
                for line in block.get('lines', []):
                    for span in line['spans']:
                        cur = []
                        for ch in span['chars'] + [None]:
                            if ch is None or ch['c'].isspace():
                                if cur:
                                    out.append({'page': pno + 1, 'text': ''.join(c['c'] for c in cur),
                                                'x': cur[0]['origin'][0], 'baseline': cur[0]['origin'][1],
                                                'ytop': cur[0]['bbox'][1], 'size': span['size'],
                                                'font': span['font']})
                                cur = []
                            else:
                                cur.append(ch)
        _cache[key] = out
    return _cache[key]


def pair_for(run, fx, k):
    text, _, occ = k.rpartition('#')
    n = 0
    for p in probe(run, fx)['pairs']:
        if unicodedata.normalize('NFC', p['text']) == text:
            n += 1
            if n == int(occ):
                return p
    return None


def same_word(pymu_text, probe_text):
    a = unicodedata.normalize('NFC', pymu_text)
    b = unicodedata.normalize('NFC', html.unescape(probe_text))
    if a == b:
        return True
    # PyMuPDF splits a word at a font change (bold "negrito" + regular ","), pdftotext does not
    return len(a) >= 2 and (b.startswith(a) and len(a) >= len(b) - 2 or a.startswith(b) and len(b) >= len(a) - 2)


def pymu(run, fx, which, side, text):
    best = None
    for w in words(run, fx, which):
        if w['page'] != side['page'] or not same_word(w['text'], text):
            continue
        d = abs(w['x'] - side['x']) + abs(w['ytop'] - side['y']) * 0.25
        if best is None or d < best[0]:
            best = (d, w)
    return None if best is None or best[0] > 6 else best[1]


def r3(v):
    return None if v is None else round(v, 3)


def layer_y_moves(run, fx):
    """(runs whose final-layer text dy == B0 dy, runs with both) for a fixture's layout-diff."""
    key = ('layer', run, fx)
    if key not in _cache:
        d = json.load(open(f'{run}/layout-diff/{fx}/run-all/diff.json'))
        final = d['final_layer']
        same = both = 0
        for ev in d['text_events']:
            pl = ev['per_layer']
            f = (pl.get(final) or {}).get('dy')
            b = (pl.get('B0') or {}).get('dy')
            if f is None or b is None:
                continue
            both += 1
            same += abs(f - b) <= 0.01
        _cache[key] = (same, both)
    return _cache[key]


def refw_stats(run):
    """REF-W tol-2 consensus distribution per fixture (oracle layout result.json)."""
    key = ('refw', run)
    if key not in _cache:
        import glob
        out = {}
        for f in glob.glob(run + '/oracles/*/*/plain/result.json'):
            name = f.split('/')[-3]
            lay = json.load(open(f))['results'].get('layout') or {}
            for t in lay.get('tolerances', []):
                if t['tol'] == 2 and t['subjectError'].get('medianDy') is not None:
                    se = t['subjectError']
                    out[name] = {'words': t['consensus'], 'medianDy': se['medianDy'],
                                 'meanAbsDy': se['meanAbsDy'], 'p50': se['p50Distance'],
                                 'p95': se['p95Distance'], 'max': se['maxDistance']}
        _cache[key] = out
    return _cache[key]


# Per fixture|criterion notes measured for this step (layout-diff of the new run; Chrome 155).
VALUE_NOTES = {
    'notification|LET-1': (' Worst anchor "Gerente do Departamento de Cobrança": raw Blitz on the source HTML '
                           '(layout-diff N0, Blitz\'s own text placement) is already +6.516 px = +4.89 pt below '
                           'Chrome, with its box at the same +6.516 px (text-in-box exact) and every later layer '
                           '(B0, L0, passes, P) equal: the new 1.81 mm (5.13 pt vs the Chrome/WeasyPrint '
                           'midpoint) is that raw Blitz drift; the old 0.75 mm was it minus the 3 pt raise.'),
    'notification|REF-W': (' The tail is the lower half of the letter, which raw Blitz places +6.5 px low (N0; '
                           'deadline table box +4.6 px with dh +1.0, signature block +6.516 px).'),
    'danfe|DANFE-L1a': ' Every DANFE-L1 label moves by exactly +1.06 mm: DANFE labels were already below Chrome.',
    'danfe|DANFE-L1b': ' Every DANFE-L1 label moves by exactly +1.06 mm: DANFE labels were already below Chrome.',
    'danfe|DANFE-L1c': ' Every DANFE-L1 label moves by exactly +1.06 mm: DANFE labels were already below Chrome.',
    'danfe|DANFE-L1d': ' Every DANFE-L1 label moves by exactly +1.06 mm: DANFE labels were already below Chrome.',
    'text-stress-ptbr|REF-W': (' The value is dominated by a pre-existing page-break displacement: vellora '
                               'paginates 8 pages vs Chrome 7 (REF-P unchanged by this step), so most rows of '
                               'pages 2-8 sit 240-280 pt below the same row in Chrome (spot check, page 3: '
                               '"10/04/2026 Compra no débito ..." at 422.8 pt vs Chrome 150.6 pt); p50 244 pt '
                               'before and after.'),
}


ENV = ("Chrome 155.0.8059.85 local macOS (INDICATIVE), 2026-10-21; old = port/03 run 20261021T114420Z, "
       "new = this run; PyMuPDF 1.28.2 span origins of layout-diff/<fixture>/run-all/{vellora,chrome}.pdf. "
       "Numbers: attachments/accepted-evidence.json; label legend: baselines/port/README.md (04-block-offset)")
EVID_SHORT = ("PyMuPDF baselines, layout-diff PDFs, port/03 run vs this run, Chrome 155 local (INDICATIVE); "
              "attachments/accepted-evidence.json")

accept, evidence, unlabelled = [], {}, []
for it in gate['items']:
    if not it.get('gating'):
        continue
    key = it['key']
    rec = {'key': key, 'before': it.get('before'), 'after': it.get('after'), 'detail': it.get('detail')}
    label = reason = None
    if key.startswith('pdf-probe|'):
        _, fx, k, axis = key.split('|')
        sides = {}
        for lab, run in (('old', old_run), ('new', new_run)):
            p = pair_for(run, fx, k)
            if p is None:
                sides[lab] = None
                continue
            v = pymu(run, fx, 'vellora', p['vellora'], p['text'])
            c = pymu(run, fx, 'chrome', p['chrome'], p['text'])
            sides[lab] = {
                'probe_dx': p['dx_pt'], 'probe_dy': p['dy_pt'],
                'vellora_page': p['vellora']['page'],
                'baseline_dy': r3(v['baseline'] - c['baseline']) if v and c else None,
                'origin_dx': r3(v['x'] - c['x']) if v and c else None,
                'font': v and v['font'], 'size': r3(v and v['size']),
            }
        rec.update(sides)
        o, n = sides.get('old'), sides.get('new')
        ok = o and n and o['baseline_dy'] is not None and n['baseline_dy'] is not None
        if o and n and abs(o['probe_dx']) > PAIR_DX_PT and abs(n['probe_dx']) > PAIR_DX_PT:
            label = 'pdf-probe-pairing-artefact'
            reason = (f'k-th-occurrence pair is another word instance: paired Chrome word '
                      f'{n["probe_dx"]:+.1f} pt away horizontally (old {o["probe_dx"]:+.1f} pt)')
        elif ok and axis == 'dy':
            rec['ascent_term_new'] = r3(n['probe_dy'] - n['baseline_dy'])
            rec['ascent_term_old'] = r3(o['probe_dy'] - o['baseline_dy'])
            grow = abs(n['baseline_dy']) - abs(o['baseline_dy'])
            rec['baseline_abs_dy_growth'] = r3(grow)
            same, both = layer_y_moves(new_run, fx)
            if grow <= TOL + 1e-9:
                label = 'pdf-probe-ascent-artefact'
                reason = (f'baseline {o["baseline_dy"]:+.2f} -> {n["baseline_dy"]:+.2f} pt vs Chrome (not worse '
                          f'beyond 1 pt); pdf-probe yMin carries +{rec["ascent_term_new"]} pt (/Ascent krilla '
                          f'sTypoAscender vs Chrome hhea, {n["size"]} pt {n["font"]})')
            else:
                shift = n['baseline_dy'] - o['baseline_dy']
                rec['baseline_shift'] = r3(shift)
                why = ('= the removed 3 pt raise' if abs(shift - 3.0) <= 0.1 else
                       f'= the removed 3 pt raise {shift - 3.0:+.2f} pt block_offset')
                if abs(o['baseline_dy']) >= MISPLACED_PT:
                    label = 'misplaced-word-uniform-shift'
                    reason = (f'already {abs(o["baseline_dy"]):.2f} pt off Chrome before; baseline '
                              f'{o["baseline_dy"]:+.2f} -> {n["baseline_dy"]:+.2f} pt, moved {shift:+.2f} pt {why}')
                else:
                    label = 'compensating-drift'
                    reason = (f'baseline {o["baseline_dy"]:+.2f} -> {n["baseline_dy"]:+.2f} pt vs Chrome, moved '
                              f'{shift:+.2f} pt {why}; the old value was the layout error minus that raise')
        elif (not ok and axis == 'dy' and o and n and abs(o['probe_dy']) >= 10.0
              and abs(n['probe_dy'] - o['probe_dy'] - 3.0) <= 0.1):
            label = 'misplaced-word-uniform-shift'
            reason = (f'already {abs(o["probe_dy"]):.2f} pt off Chrome before (pdftotext yMin; PyMuPDF does not '
                      f'resolve the word); {o["probe_dy"]:+.2f} -> {n["probe_dy"]:+.2f} pt = the removed 3 pt raise')
        elif ok and axis == 'dx':
            grow = abs(n['origin_dx']) - abs(o['origin_dx'])
            rec['origin_abs_dx_growth'] = r3(grow)
            if grow <= TOL + 1e-9:
                label = 'pdf-probe-ascent-artefact'
                reason = (f'dx item: the pen origin (PyMuPDF) vs Chrome went {o["origin_dx"]:+.2f} -> '
                          f'{n["origin_dx"]:+.2f} pt; pdf-probe paired a different word occurrence.')
    elif key.startswith('code|') or key.startswith('canary|') or key.startswith('layout-diff|'):
        unlabelled.append(key)  # absolute checks are never accepted
        continue
    elif key.startswith('wpt|'):
        label = 'wpt-subpixel-raster-artefact'
        reason = ('Lost to 1 differing pixel (maxDifference 3 of 255) at a glyph-run boundary: the test '
                  'draws three inline runs, the reference one. Chrome 155 FAILS the same reftest in the '
                  'same harness (78 px, maxDifference 64; reused Chrome results). The old pass drew both '
                  'pages 4 px above Chrome (ink bbox top 14 vs Chrome 19; now 18).')
    else:
        # value / status rows of the corpus and oracles (REF-W, LET-1, DANFE-L1, INV-R1, ...)
        fx = key.split('|')[0]
        label = 'compensating-drift'
        b, a = it.get('before'), it.get('after')
        delta = (a - b) if isinstance(a, (int, float)) and isinstance(b, (int, float)) else None
        rec['delta'] = r3(delta)
        so, sn = refw_stats(old_run).get(fx), refw_stats(new_run).get(fx)
        rec['refw_tol2_old'], rec['refw_tol2_new'] = so, sn
        body = ''
        if so and sn:
            body = (f' REF-W distribution over the {sn["words"]} consensus words (pt, old -> new): median dy '
                    f'{so["medianDy"]:+.2f} -> {sn["medianDy"]:+.2f}, mean |dy| {so["meanAbsDy"]:.2f} -> '
                    f'{sn["meanAbsDy"]:.2f}, p50 {so["p50"]:.2f} -> {sn["p50"]:.2f}, p95 {so["p95"]:.2f} -> '
                    f'{sn["p95"]:.2f}, max {so["max"]:.2f} -> {sn["max"]:.2f}.')
        extra = VALUE_NOTES.get(key.split('|')[0] + '|' + key.split('|')[1], '')
        moved = (f'{b} -> {a}' + (f' ({delta:+.2f} mm; the removed raise is 3 pt = 1.06 mm)' if delta is not None else ''))
        reason = (f'{moved}. The anchor/tail words now sit at their laid-out and paginated baseline (pdf.rs adds nothing), '
                  f'whose error vs the reference is raw Blitz box drift that the constant 3 pt raise had '
                  f'offset.{extra}{body}')
    if label is None:
        unlabelled.append(key)
        continue
    rec['label'] = label
    evidence[key] = rec
    accept.append({'key': key, 'after': it.get('after'), 'label': label, 'reason': reason,
                   'evidence': (ENV if not key.startswith('pdf-probe|') else EVID_SHORT)})

if unlabelled:
    sys.stderr.write('UNLABELLED %d\n' % len(unlabelled))
    for k in unlabelled[:50]:
        sys.stderr.write('  %s\n' % k)
json.dump({'acceptedRegressions': accept}, open(out_accept, 'w'), indent=1, ensure_ascii=False)
json.dump(evidence, open(out_evidence, 'w'), separators=(',', ':'), ensure_ascii=False)
import collections
print(collections.Counter(a['label'] for a in accept), 'unlabelled', len(unlabelled))
