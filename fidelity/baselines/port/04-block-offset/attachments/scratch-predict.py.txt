"""Predict GATE-P1 (pdf-probe dy) outcome of a uniform +3 pt baseline shift from a run's pdf-probe."""
import json, sys, glob, collections

root = sys.argv[1]
shift = float(sys.argv[2]) if len(sys.argv) > 2 else 3.0
tot = collections.Counter()
for f in sorted(glob.glob(root + '/layout-diff/*/run-all/pdf-probe.json')):
    d = json.load(open(f))
    name = f.split('/')[-3]
    c = collections.Counter()
    for p in d['pairs']:
        if not p.get('samePage'):
            continue
        dy = p['dy_pt']
        new = dy + shift
        if abs(new) > abs(dy) + 1.0:
            c['regress'] += 1
        elif abs(new) < abs(dy) - 1.0:
            c['improve'] += 1
        else:
            c['within'] += 1
    tot.update(c)
    print(f'{name:24s} pairs {len(d["pairs"]):4d}  {dict(c)}')
print('total', dict(tot))
