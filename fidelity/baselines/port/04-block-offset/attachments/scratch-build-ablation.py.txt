def refw(run):
    out = {}
    for f in glob.glob(run + '/oracles/*/*/plain/result.json'):
        name = f.split('/')[-3]
        lay = json.load(open(f))['results'].get('layout') or {}
        for t in lay.get('tolerances', []):
            if t['tol'] == 2 and t['subjectError'].get('medianDy') is not None:
                se = t['subjectError']
                out[name] = {k: round(se[v], 2) for k, v in (('medianDy', 'medianDy'), ('meanAbsDy', 'meanAbsDy'),
                             ('p50', 'p50Distance'), ('p95', 'p95Distance'), ('max', 'maxDistance'))}
                out[name]['words'] = t['consensus']
    return dict(sorted(out.items()))


def wpt(run):
    d = json.load(open(run + '/wpt/results.json'))
    passes = sum(1 for t in d['tests'] if t['outcome'] == 'pass')
    return {'tests': len(d['tests']), 'pass': passes}
