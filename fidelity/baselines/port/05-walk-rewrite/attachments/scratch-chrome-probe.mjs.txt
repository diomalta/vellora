// Scratch probe (not part of the repo): Chrome's border boxes and text line rects of every
// element with an id, for the step-05 walk probes, laid out by the ONE shared reference setup
// (fidelity/lib/chrome-reference.mjs: print media, bundled Liberation/DejaVu @font-face block).
// Viewport width = the @page content width vellora hands Blitz, truncated to whole px
// (A4 - 2 x 18 mm = 657.64 -> 657). Also prints each probe to PDF with the shared print path.
//   node chrome-probe.mjs <repo> probes.json <outDir> [viewportWidth] > chrome.json
import { mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { pathToFileURL } from "node:url";

const [repo, probesFile, outDir, vw = "657"] = process.argv.slice(2);
const ref = await import(pathToFileURL(join(repo, "fidelity/lib/chrome-reference.mjs")).href);
const probes = JSON.parse(readFileSync(probesFile, "utf8"));
mkdirSync(outDir, { recursive: true });
const chrome = await ref.launchReferenceChrome();
const width = Number(vw);
const out = {
  chrome: chrome.version,
  userAgent: chrome.userAgent,
  viewport: [width, 1000],
  printOptions: ref.PRINT_OPTIONS,
  probes: {},
};
try {
  for (const [name, html] of Object.entries(probes)) {
    const full = ref.referenceHtml(html);
    out.probes[name] = await ref.withPrintPage(chrome, full, async (page) => {
      await page.setViewport({ width, height: 1000, deviceScaleFactor: 1 });
      return page.evaluate(() => {
        const r = {};
        const lineRects = (root) => {
          const lines = [];
          const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT);
          for (let t = walker.nextNode(); t; t = walker.nextNode()) {
            if (!t.textContent.trim()) continue;
            const range = document.createRange();
            range.selectNodeContents(t);
            const rects = [...range.getClientRects()];
            // Baseline of the text node's first line: a zero-size inline-block marker inserted
            // right before it sits on the baseline (CSS 2.2 §10.8.1); removed again.
            const marker = document.createElement("span");
            marker.style.cssText = "display:inline-block;width:0;height:0;margin:0;padding:0;border:0";
            t.parentNode.insertBefore(marker, t);
            const baseline = marker.getBoundingClientRect().bottom;
            marker.remove();
            rects.forEach((q, i) => {
              lines.push({
                text: t.textContent.trim().slice(0, 40),
                x: q.x,
                y: q.y,
                w: q.width,
                h: q.height,
                ...(i === 0 ? { baseline } : {}),
              });
            });
          }
          return lines;
        };
        for (const el of [document.body, ...document.querySelectorAll("[id]")]) {
          const b = el.getBoundingClientRect();
          const cs = getComputedStyle(el);
          r[el.id || "BODY"] = {
            x: b.x,
            y: b.y,
            w: b.width,
            h: b.height,
            font: `${cs.fontWeight} ${cs.fontSize}/${cs.lineHeight} ${cs.fontFamily}`,
            color: cs.color,
            textAlign: cs.textAlign,
            lines: lineRects(el),
          };
        }
        return r;
      });
    });
    const pdf = await ref.printPdf(chrome, full);
    writeFileSync(join(outDir, `${name}.chrome.pdf`), pdf);
  }
} finally {
  await chrome.close();
}
console.log(JSON.stringify(out, null, 1));
