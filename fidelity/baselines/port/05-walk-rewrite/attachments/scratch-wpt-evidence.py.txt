"""Scratch: per-test WPT evidence (old run, new run, Chrome) for accepted WPT regressions.
  python3 -I wpt-evidence.py <old results.json> <new results.json> <test> [...] > out.json
"""
import json
import sys

old = {t["test"]: t for t in json.load(open(sys.argv[1]))["tests"]}
new = {t["test"]: t for t in json.load(open(sys.argv[2]))["tests"]}
out = {}
for k in sys.argv[3:]:
    o, n = old.get(k, {}), new.get(k, {})
    ch = o.get("chrome") or n.get("chrome") or {}
    out[k] = {
        "old": {
            "outcome": o.get("outcome"),
            "comparisons": [{x: c.get(x) for x in ("rhs", "pass", "maxDifference", "totalPixels", "pageCounts")} for c in o.get("comparisons", [])],
            "ink": o.get("ink"),
            "inkCheck": o.get("inkCheck"),
        },
        "new": {
            "outcome": n.get("outcome"),
            "comparisons": [{x: c.get(x) for x in ("rhs", "pass", "maxDifference", "totalPixels", "pageCounts")} for c in n.get("comparisons", [])],
            "ink": n.get("ink"),
            "inkCheck": n.get("inkCheck"),
        },
        "chrome": {
            "outcome": ch.get("outcome"),
            "ink": ch.get("ink"),
            "comparisons": [{x: c.get(x) for x in ("pass", "maxDifference", "pageCounts")} for c in ch.get("comparisons", [])],
        },
    }
json.dump(out, sys.stdout, indent=1)
