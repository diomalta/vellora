"""Scratch: assemble attachments/chrome-probes-walk.json for the step-05 baseline.
  fidelity/.venv/bin/python -I build-probe-attachment.py probes.json chrome.json chrome-pdf-dir out.json
"""
import json
import os
import sys

import pymupdf as fitz

probes_file, chrome_file, pdf_dir, out = sys.argv[1:5]
probes = json.load(open(probes_file))
chrome = json.load(open(chrome_file))


def pdf_words(path):
    doc = fitz.open(path)
    words = []
    for block in doc[0].get_text("rawdict")["blocks"]:
        for line in block.get("lines", []):
            for span in line["spans"]:
                text = "".join(c["c"] for c in span["chars"]).strip()
                if text:
                    x, y = span["origin"]
                    words.append({
                        "text": text,
                        "origin_pt": [round(x, 3), round(y, 3)],
                        # Chrome prints the 18 mm page margin at 68 CSS px = 51.0 pt.
                        "origin_css_px": [round((x - 51.0) / 0.75, 3), round((y - 51.0) / 0.75, 3)],
                        "size_pt": round(span["size"], 3),
                    })
    return words


result = {
    "what": "Chrome reference numbers of the step-05 walk probes (crates/vellora-core/tests/walk.rs)",
    "date": "2026-10-21",
    "environment": "local macOS 27.0.1 arm64, Chrome 155.0.8059.85 (INDICATIVE)",
    "setup": "fidelity/lib/chrome-reference.mjs: print media, bundled Liberation/DejaVu @font-face, viewport 657 x 1000 (A4 - 2 x 18 mm, truncated like vellora's)",
    "method": {
        "boxes": "getBoundingClientRect() of every element with an id (and body), CSS px",
        "lines": "Range.getClientRects() of each text node; baseline = bottom of a zero-size inline-block inserted right before the text node",
        "pdf": "PyMuPDF span origins of page.pdf(PRINT_OPTIONS), page 1",
    },
    "chrome": chrome["chrome"],
    "probes": {},
}
for name, html in probes.items():
    entry = {"html": html, "dom": chrome["probes"][name]}
    pdf = os.path.join(pdf_dir, f"{name}.chrome.pdf")
    if os.path.exists(pdf):
        entry["pdf_words"] = pdf_words(pdf)
    result["probes"][name] = entry
json.dump(result, open(out, "w"), indent=1, ensure_ascii=False)
print("wrote", out, len(result["probes"]), "probes")
