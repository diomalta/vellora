"""Final-layer text runs and painted boxes of layout_dump JSON files (old-code evidence).
  python3 -I oldruns.py dump.json [...]
"""
import json, sys

for f in sys.argv[1:]:
    d = json.load(open(f))
    layer = d["layers"][-1]
    bf, rf = layer["box_fields"].split(","), layer["run_fields"].split(",")
    boxes = [dict(zip(bf, b)) for b in layer["boxes"]]
    paths = {n["id"]: n["path"] for n in d["nodes"]}
    texts = {(m["box"], m["run"]): m["text"] for m in d["runs"]}
    print("==", f)
    for r in layer["runs"]:
        rr = dict(zip(rf, r))
        bi, ri = rr["box_index"], rr["run_index"]
        print(f"  run {texts.get((bi, ri))!r:<30} ({rr['origin_x']},{rr['origin_y']}) owner {paths.get(boxes[bi]['node_id'], '?').split('>')[-1]}")
    for b in boxes:
        p = paths.get(b["node_id"], "?")
        if any(k in p for k in ("abs", "z", "br", "span", "div:nth-child(3)", "div:nth-child(4)")):
            print(f"  box {p.split('>')[-1]:<28} ({b['x']},{b['y']} {b['w']}x{b['h']})")
