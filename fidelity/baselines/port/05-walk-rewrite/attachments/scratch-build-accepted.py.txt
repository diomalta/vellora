"""Scratch (not part of the repo): build accepted.json for port step 05 from the gate report.
  python3 -I build-accepted.py <gate-report.json> <wpt-accepted-evidence.json> <out accepted.json> <out evidence.json>
Every gating item gets one label; only `wpt-false-pass` fits GATE-A1's literal wording.
"""
import json
import sys

report = json.load(open(sys.argv[1]))
wpt_ev = json.load(open(sys.argv[2]))
OLD = "port/04 run 20261021T125346Z (8e4face)"
NEW = "port/05 run 20261021T142201Z (e9dc7e4)"
ENV = "Chrome 155.0.8059.85 local macOS (INDICATIVE), 2026-10-21"
EVID = (
    f"{ENV}; old = {OLD}, new = {NEW}; WPT test/ref/Chrome records of wpt/results.json "
    "(Chrome outcomes and ink reused from run-all-beta4-20261021T054809Z). Numbers: "
    "attachments/accepted-evidence.json; label legend: baselines/port/README.md (05-walk-rewrite)"
)


def ink(k, side):
    v = wpt_ev[k][side]
    i = v["ink"] or {}
    pc = v["comparisons"][0]["pageCounts"] if v["comparisons"] else None
    return f"ink bbox {i.get('bbox')} ({i.get('px')} px, pages {pc})"


def chrome_ink(k):
    c = wpt_ev[k]["chrome"]
    pc = c["comparisons"][0]["pageCounts"] if c["comparisons"] else None
    return f"ink bbox {c['ink']['bbox']} ({c['ink']['px']} px, pages {pc})"


FALSE_PASS = {
    "css/CSS2/floats-clear/floats-153.xht": (
        "The green PASS (a zero-width right float with margin-left -10em in a padded cell) is in "
        "Chrome's ink and in neither old render: Blitz places the float at x -70 (raw Blitz, "
        "beta.4; unchanged by this step), and the reference's PASS (an inline relative <div>, "
        "text in an anonymous block of <body>) was never read by the DOM walk. Test and "
        "reference were the same blank-but-instructions picture. Now the reference draws PASS "
        "(walk reads the anonymous block), the test still does not (Blitz float placement)."
    ),
    "css/CSS2/floats/floats-placement-vertical-003.xht": (
        "Old test and reference drew the same picture 29 px off Chrome's: the yellow squares "
        "at y 142-172 instead of 113-143 (same colour bins, only position differs). "
        "html_normalize had wrapped the leading 'H' in a <div>, so both documents were block "
        "containers whose floats Blitz (no `floats` feature) stacked as blocks."
    ),
    "css/CSS2/floats/floats-placement-vertical-004.xht": (
        "Old test and reference drew the same picture, unlike Chrome's: Chrome puts the second "
        "blue float beside the first (ink to x 207, bottom 113), the old render stacked them "
        "(x 107, bottom 212); bboxShift exactly 100 px, one below the ink-displaced flag "
        "threshold (> 100)."
    ),
    "css/CSS2/linebox/baseline-block-with-overflow-001.html": (
        "Old test and reference drew the same wrong picture: html_normalize wrapped each 'XX' "
        "next to the inline-block .outer <div> (block-like by tag) in a <div>, putting the text "
        "on its own line (ink 37 px wide, 415 px tall; Chrome 64 x 329). The test now draws "
        "Chrome's geometry exactly (ink bbox [8, 8, 64, 329] = Chrome's, 10090 vs 10110 px); "
        "the reference does not."
    ),
    "css/css-break/flexbox/flex-fragmented-with-float-descendant-001.html": (
        "Old test and reference were equal only because vellora ignores both multicol "
        "(`columns`) and floats (blitz-dom `floats` feature off): the test's float:left .float "
        "rendered as the reference's plain block. Old render: one column, the 500 px float "
        "drawn whole below/over the 160 px multicol box (64866 px of ink, bbox bottom 576); "
        "Chrome fragments it into columns (42826 px, different colour bins: old has bins 63 = "
        "2500 px and 260 = 900 px Chrome lacks, Chrome has bin 7 = 5488 px the old lacks). "
        "Pictures: attachments/flex-fragmented-old-{test,ref}.png."
    ),
    "css/css-page/monolithic-overflow-014-print.html": (
        "Old test and reference printed 2 pages each; Chrome prints 4 pages for both (the 350vh "
        "monolithic inline-block). Same wrong picture on both sides (92160 px of ink vs Chrome's "
        "73728)."
    ),
}
COMPENSATION_WPT = {
    "css/CSS2/floats-clear/floats-132.xht": "a left float followed by text inside div.box",
    "css/css-text/text-indent/below-float.html": "a left float inside a <span> followed by text",
    "css/css-text/text-indent/below-float2.html": "a left float inside a <span> followed by text",
}

accepted = []


def ext2(path):
    d = json.load(open(path))
    return {
        e["name"]: {
            "missing": e.get("subjectOnly", {}).get("missing"),
            "extra": e.get("subjectOnly", {}).get("extra"),
        }
        for e in d["results"]["text"]["extractors"]
    }


evidence = {
    "wpt": wpt_ev,
    "postmark": {
        "root_cause": "blitz-html 0.3.0-beta.4 html_sink.rs parse_into_mutator: '<!DOCTYPE' + 'XHTML' on the first line -> parse_xml_into_mutator; the template has 3 <br> (lines 545, 584, 585 of templated.html), each swallows the text after it",
        "ext2_old": ext2(sys.argv[5]),
        "ext2_new": ext2(sys.argv[6]),
        "pdftotext_layout_old_tail": open(sys.argv[7]).read().splitlines()[-8:],
        "pdftotext_layout_new_tail": open(sys.argv[8]).read().splitlines()[-8:],
    },
}
for item in report["items"]:
    if not item.get("gating"):
        continue
    key = item["key"]
    entry = {"key": key, "after": item["after"]}
    if key.startswith("wpt|"):
        t = key[4:]
        if t in FALSE_PASS:
            entry["label"] = "wpt-false-pass"
            entry["reason"] = (
                f"pass -> fail. FALSE PASS: the old test matched its reference (maxDifference 0) "
                f"but not Chrome's rendering of the test: old {ink(t, 'old')}, Chrome "
                f"{chrome_ink(t)}. {FALSE_PASS[t]} New: {ink(t, 'new')}."
            )
        elif t in COMPENSATION_WPT:
            entry["label"] = "html-normalize-compensation"
            entry["reason"] = (
                f"pass -> fail. Does NOT fit GATE-A1's literal wording: the old pass matched "
                f"Chrome (old {ink(t, 'old')}, Chrome {chrome_ink(t)}). It came from "
                f"html_normalize rewriting the DOM ({COMPENSATION_WPT[t]}: the text was wrapped "
                f"in an inserted <div>, turning the inline formatting context into a block "
                f"container). Without the rewrite Blitz 0.3.0-beta.4 lays the float out inside "
                f"the inline formatting context and ignores it for the line (floats-132: float "
                f"at location (0, -4) under unwrapped text); the box walk reads that faithfully. "
                f"Enabling blitz-dom's `floats` feature does NOT restore it (CONFIRMED, step-05 "
                f"WPT experiment). New: {ink(t, 'new')}."
            )
        else:
            raise SystemExit(f"unlabelled WPT item {key}")
    elif key.startswith("oss-postmark-receipt|"):
        entry["label"] = "html-normalize-compensation"
        entry["reason"] = (
            f"{item['before']} -> {item['after']}. Does NOT fit GATE-A1's literal wording; a real "
            "fidelity loss on this document class. The template's first line is an XHTML 1.0 "
            "Transitional doctype, so blitz-html 0.3.0-beta.4 parses it with its XML parser "
            "(src/html_sink.rs:92-99), where <br> is not void: the text after each of the 3 <br> "
            "becomes the <br>'s content and is not rendered ('The [Product Name] team', "
            "'1234 Street Rd.', 'Suite 1234'). Until step 04 html_normalize happened to change "
            "this document (1 wrapper), re-serialized it and Blitz re-parsed the serialization "
            "as HTML (Chrome parses text/html as HTML). The root cause has been documented in "
            "the layout-diff allowlist since port/00-start (source_dom_keys_vs_chromium); the "
            "parse-mode decision (HTML for every text/html input, with the WPT .xht trade-off) "
            "is a follow-up. Also changed: 'Download as PDF' is now drawn (it was missing)."
        )
    else:
        raise SystemExit(f"unlabelled item {key}")
    entry["evidence"] = EVID
    accepted.append(entry)
json.dump({"acceptedRegressions": accepted}, open(sys.argv[3], "w"), indent=1, ensure_ascii=False)
json.dump(evidence, open(sys.argv[4], "w"), indent=1, ensure_ascii=False)
from collections import Counter

print(len(accepted), Counter(e["label"] for e in accepted))
