// Scratch probe used in port step 05 (2026-10-21), NOT part of the repo. It was appended
// temporarily to crates/vellora-core/src/blitz_engine.rs (the only module allowed to touch Blitz)
// and driven by a scratch example; both were removed before the step's commit.
//
//   // crates/vellora-core/examples/scratch_probe.rs
//   fn main() {
//       let args: Vec<String> = std::env::args().collect();
//       let html = std::fs::read_to_string(&args[1]).unwrap();
//       let width: f64 = args.get(2).map(|s| s.parse().unwrap()).unwrap_or(657.0);
//       print!("{}", vellora_core::blitz_engine::scratch_probe_tree(&html, width, false));
//   }

#[doc(hidden)]
pub fn scratch_probe_tree(html: &str, width: f64, normalize: bool) -> String {
    use std::fmt::Write as _;
    let _ = normalize;
    let mut doc = build_document(html, &[]);
    let viewport = Viewport::new(width as u32, 1000, 1.0, ColorScheme::Light);
    doc.as_mut().set_viewport(viewport);
    doc.as_mut().resolve(0.0);
    let base: &BaseDocument = doc.as_ref();
    let mut out = String::new();
    // Origins summed down DOM children.
    let mut dom_pos: std::collections::HashMap<NodeId, (f32, f32)> = Default::default();
    let mut st = vec![(base.root_node().id, 0f32, 0f32)];
    while let Some((id, px, py)) = st.pop() {
        let n = base.get_node(id).unwrap();
        let (x, y) = if node_has_layout_data(n) {
            let l = n.final_layout().location;
            (px + l.x, py + l.y)
        } else {
            (px, py)
        };
        dom_pos.insert(id, (x, y));
        for &c in n.children.iter() {
            st.push((c, x, y));
        }
    }
    // Origins summed down paint_children, plus stacking-context hoisted children at
    // HoistedPaintChild::position (blitz-paint draw_children).
    let mut paint_pos: std::collections::HashMap<NodeId, (f32, f32)> = Default::default();
    let mut st = vec![(base.root_element().id, 0f32, 0f32)];
    while let Some((id, px, py)) = st.pop() {
        let n = base.get_node(id).unwrap();
        let l = n.final_layout().location;
        let (x, y) = (px + l.x, py + l.y);
        paint_pos.insert(id, (x, y));
        for &c in n.paint_children.borrow().iter().flatten() {
            st.push((c, x, y));
        }
        if let Some(sc) = &n.stacking_context {
            for h in sc.children.iter() {
                let p = h.position(n.tree(), n.id);
                st.push((h.node_id, x + p.x, y + p.y));
            }
        }
    }
    // Origins summed down layout_children.
    let mut layout_pos: std::collections::HashMap<NodeId, (f32, f32)> = Default::default();
    let mut st = vec![(base.root_element().id, 0f32, 0f32)];
    while let Some((id, px, py)) = st.pop() {
        let n = base.get_node(id).unwrap();
        if !node_has_layout_data(n) {
            continue;
        }
        let l = n.final_layout().location;
        let (x, y) = (px + l.x, py + l.y);
        layout_pos.insert(id, (x, y));
        for &c in n.layout_children.borrow().iter().flatten() {
            st.push((c, x, y));
        }
    }
    // Print every node reachable through DOM children, ::before/::after and layout_children.
    let mut seen = std::collections::HashSet::new();
    let mut st = vec![(base.root_node().id, 0usize)];
    while let Some((id, depth)) = st.pop() {
        if !seen.insert(id) {
            continue;
        }
        let n = base.get_node(id).unwrap();
        let kind = if n.is_anonymous() {
            "anon"
        } else if n.is_element() {
            "el"
        } else if n.is_text_node() {
            "text"
        } else {
            "other"
        };
        let tag = n
            .data
            .downcast_element()
            .map(|e| e.name.local.to_string())
            .unwrap_or_default();
        let cls = n.attr(blitz_dom::local_name!("class")).unwrap_or("");
        let idattr = n.attr(blitz_dom::local_name!("id")).unwrap_or("");
        let ids = |v: &[NodeId]| {
            v.iter()
                .map(|i| i.as_u64().to_string())
                .collect::<Vec<_>>()
                .join(",")
        };
        let lc: Option<Vec<NodeId>> = n
            .layout_children
            .borrow()
            .as_ref()
            .map(|c| c.iter().copied().collect());
        let pc: Option<Vec<NodeId>> = n
            .paint_children
            .borrow()
            .as_ref()
            .map(|c| c.iter().copied().collect());
        let sc: Option<Vec<NodeId>> = n
            .stacking_context
            .as_ref()
            .map(|s| s.children.iter().map(|h| h.node_id).collect());
        let ind = "  ".repeat(depth);
        let _ = write!(
            out,
            "{ind}#{} {kind} <{tag}> id={idattr:?} class={cls:?} parent={:?} lp={:?} oofcb={:?}",
            id.as_u64(),
            n.parent.map(|p| p.as_u64()),
            n.layout_parent.get().map(|p| p.as_u64()),
            n.oof_containing_block.get().map(|p| p.as_u64())
        );
        if node_has_layout_data(n) {
            let l = n.final_layout();
            let ap = n.absolute_position(0.0, 0.0);
            let _ = write!(
                out,
                " loc=({},{}) size=({}x{}) m=({},{}) abs=({},{}) dom={:?} paint={:?} layout={:?} disp={:?} pos={:?} z={}",
                l.location.x, l.location.y, l.size.width, l.size.height, l.margin.top, l.margin.bottom,
                ap.x, ap.y, dom_pos.get(&id), paint_pos.get(&id), layout_pos.get(&id),
                n.primary_styles().map(|s| format!("{:?}/{:?}", s.clone_display().outside(), s.clone_display().inside())),
                n.primary_styles().map(|s| s.clone_position()),
                n.z_index()
            );
        }
        if n.flags.is_inline_root() {
            if let Some(t) = n.data.downcast_element().and_then(|e| e.inline_layout_data.as_ref()) {
                let _ = write!(out, " INLINE_ROOT text={:?} bo={}", t.text.chars().take(60).collect::<String>(), t.block_offset);
            }
        }
        if let Some(t) = n.text_data() {
            let _ = write!(out, " text={:?}", t.content.chars().take(40).collect::<String>());
        }
        let _ = writeln!(
            out,
            "\n{ind}   dom=[{}] before={:?} after={:?} lc={:?} pc={:?} sc={:?}",
            ids(&n.children),
            n.before().map(|p| p.as_u64()),
            n.after().map(|p| p.as_u64()),
            lc.as_deref().map(ids),
            pc.as_deref().map(ids),
            sc.as_deref().map(ids)
        );
        let mut kids: Vec<NodeId> = Vec::new();
        kids.extend(n.children.iter().copied());
        kids.extend(n.before());
        kids.extend(n.after());
        kids.extend(lc.iter().flatten().copied());
        for k in kids.into_iter().rev() {
            st.push((k, depth + 1));
        }
    }
    out
}
