"""Scratch: write the step-05 walk probe documents to probes.json (name -> full HTML)."""
import json
import sys

HEAD = """<!DOCTYPE html><html><head><style>
@page { size: A4; margin: 18mm; }
body { margin: 8px; font: 16px/20px "Liberation Sans"; color: #000; }
%s
</style></head><body>%s</body></html>"""

probes = {
    # Out-of-flow box nested in a non-positioned <p> inside a relative container: its containing
    # block is the container, not the DOM parent chain.
    "abs_nested": HEAD % (
        ".rel { position: relative; margin-top: 30px; padding: 10px; background: #eee; }\n"
        ".abs { position: absolute; top: 5px; left: 40px; width: 50px; height: 20px; background: #0a0; }",
        '<div class="rel" id="rel"><p id="p">para <span class="abs" id="abs">x</span></p></div>'
        '<div id="after">after</div>',
    ),
    # Absolutely positioned box with no positioned ancestor: initial containing block. The body
    # margin and the preceding flow must not be added.
    "abs_icb": HEAD % (
        "h1 { margin: 20px 0; font-size: 20px; line-height: 24px; }\n"
        ".abs2 { position: absolute; top: 100px; left: 200px; width: 30px; height: 30px; background: #00f; }\n"
        ".z { position: absolute; z-index: 2; top: 10px; left: 300px; width: 10px; height: 10px; background: #f00; }",
        '<h1 id="h">Title</h1><div id="d">flow</div><div class="abs2" id="abs2"></div>'
        '<div class="z" id="z"></div>',
    ),
    # Text next to a block child: Blitz wraps each inline run in an anonymous block.
    "anon_blocks": HEAD % (
        "p { margin: 16px 0; }",
        '<div id="mixed">leading text<p id="mp">inner para</p>trailing <b id="b">bold</b></div>',
    ),
    # The body's own inline text after a block child (an anonymous block of the body).
    "body_text": HEAD % (
        "",
        '<div id="d">block</div>body text',
    ),
    # A body whose in-flow content is all inline-level: the body itself is the inline root.
    "body_inline_root": HEAD % (
        ".ib { display: inline-block; width: 20px; height: 10px; background: #0a0; }\n"
        ".abs { position: absolute; top: 60px; left: 100px; width: 10px; height: 10px; background: #00f; }",
        'only inline <b id="b">text</b> <span class="ib" id="ib"></span><div class="abs" id="abs"></div>',
    ),
    # Block ::before generated content.
    "before_block": HEAD % (
        ".mix::before { content: \"BEFORE\"; display: block; background: #ff0; }\n"
        ".mix::after { content: \"AFTER\"; }",
        '<div class="mix" id="mix">mix content</div><div id="next">next</div>',
    ),
    # Block ::before with block siblings (classify as block container).
    "before_block_sibling": HEAD % (
        ".mix::before { content: \"BEFORE\"; display: block; background: #ff0; }",
        '<div class="mix" id="mix"><div id="inner">inner block</div></div><div id="next">next</div>',
    ),
    # Table cell mixing text and a block child (html_normalize's original target).
    "cell_mixed": HEAD % (
        "table { border-spacing: 0; } td { padding: 8px; }",
        '<table id="t"><tr><td id="td">cell <div id="blk">block</div> after</td></tr></table>',
    ),
    # The receipt's .thanks: a display:block span inside <p>, then the note text. html_normalize
    # wrapped the note in a <div> inside the <p>; the HTML re-parse hoisted it out of the <p>.
    "thanks": HEAD % (
        ".wrap { width: 400px; }\n"
        ".thanks { text-align: center; font-size: 10pt; color: #444; margin-top: 8mm; }\n"
        ".thanks .big { display: block; font-size: 13pt; font-weight: 700; color: #1a1a1a; margin-bottom: 2mm; }\n"
        "footer { margin-top: 9mm; font-size: 8pt; }",
        '<div class="wrap" id="wrap"><p class="thanks" id="thanks">\n'
        '  <span class="big" id="big">Obrigado pela sua compra!</span>\n'
        '  Volte sempre.\n</p>\n<footer id="foot">Fine print</footer></div>',
    ),
    # Relative-positioned block with an absolutely positioned child at bottom/right.
    "abs_bottom_right": HEAD % (
        ".box { position: relative; width: 300px; height: 120px; margin: 10px 20px; border: 2px solid #000; }\n"
        ".br { position: absolute; right: 10px; bottom: 5px; width: 40px; height: 20px; background: #0a0; }",
        '<div class="box" id="box"><div id="inflow">in flow</div><div class="br" id="br">br</div></div>',
    ),
}
json.dump(probes, open(sys.argv[1], "w"), indent=1)
print(len(probes), "probes")
