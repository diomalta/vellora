// Scratch (not part of the repo): compare two shim-recorded WPT runs.
//   node compare-wpt.mjs <before run dir> <after run dir> <capture-before> <capture-after> <dump-effects.json> <out.json>
// - per-test outcome and clean/flagged changes (flag = inkCheck suspect, as the gate defines it)
// - per input key: PDF sha256 before vs after; asserts changed keys are a subset of the keys where
//   at least one deleted pass fired (from the pre-deletion dump map)
import { readFileSync, readdirSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { suspectFlags } from "../../wpt/lib/report.mjs";

const [beforeDir, afterDir, capB, capA, effectsFile, outFile] = process.argv.slice(2);
const DELETED = [
  "vertical_margin_collapse",
  "collapsed_table_medium_border_insets",
  "collapsed_table_medium_border_insets_x",
];
const load = (d) => JSON.parse(readFileSync(join(d, "results.json"), "utf8"));
const flags = (t) => (t.outcome === "pass" ? suspectFlags(t) : []);
const B = new Map(load(beforeDir).tests.map((t) => [t.test, t]));
const A = new Map(load(afterDir).tests.map((t) => [t.test, t]));

function hashes(cap) {
  const m = new Map();
  for (const f of readdirSync(cap).filter((x) => x.startsWith("log-"))) {
    for (const line of readFileSync(join(cap, f), "utf8").split("\n")) {
      if (!line) continue;
      const r = JSON.parse(line);
      const v = r.pdf ?? `error:${r.error}`;
      const prev = m.get(r.key);
      if (prev && prev !== v) m.set(r.key, `NONDETERMINISTIC ${prev} vs ${v}`);
      else m.set(r.key, v);
    }
  }
  return m;
}
const hb = hashes(capB);
const ha = hashes(capA);
const effects = JSON.parse(readFileSync(effectsFile, "utf8"));
const fired = new Map();
for (const r of effects.results) {
  if (r.status !== "ok") continue;
  const which = DELETED.filter((p) => r.effects[p] && (r.effects[p].boxes || r.effects[p].runs));
  if (which.length) fired.set(r.key, which);
}
const urlOf = (key) =>
  JSON.parse(readFileSync(join(capB, "inputs", key, "meta.json"), "utf8")).baseUrl.replace(
    "http://web-platform.test:8000/",
    "",
  );

const changedKeys = [];
const unexplained = [];
let sameKeys = 0;
let missingAfter = 0;
const nondet = [];
for (const [k, v] of hb) {
  if (v.startsWith("NONDETERMINISTIC")) nondet.push({ key: k, side: "before" });
  const w = ha.get(k);
  if (w === undefined) {
    missingAfter++;
    continue;
  }
  if (w.startsWith("NONDETERMINISTIC")) nondet.push({ key: k, side: "after" });
  if (w === v) sameKeys++;
  else {
    const entry = { key: k, url: urlOf(k), fired: fired.get(k) ?? [] };
    changedKeys.push(entry);
    if (!fired.has(k)) unexplained.push(entry);
  }
}
const firedUnchanged = [...fired.keys()].filter((k) => hb.get(k) === ha.get(k));

const outcomeChanges = [];
for (const [name, b] of B) {
  const a = A.get(name);
  const fb = flags(b);
  const fa = flags(a);
  const ob = b.outcome;
  const oa = a?.outcome;
  if (ob !== oa || (ob === "pass" && fb.length !== fa.length)) {
    outcomeChanges.push({
      test: name,
      before: `${ob}${ob === "pass" ? (fb.length ? "(flagged)" : "(clean)") : ""}`,
      after: `${oa}${oa === "pass" ? (fa.length ? "(flagged)" : "(clean)") : ""}`,
      inkBefore: b.inkCheck ?? null,
      inkAfter: a?.inkCheck ?? null,
      cmpBefore: b.comparisons?.map((c) => ({ pass: c.pass, maxDifference: c.maxDifference, totalPixels: c.totalPixels })),
      cmpAfter: a?.comparisons?.map((c) => ({ pass: c.pass, maxDifference: c.maxDifference, totalPixels: c.totalPixels })),
    });
  }
}
const result = {
  inputsBefore: hb.size,
  inputsAfter: ha.size,
  missingAfter,
  sameKeys,
  changedKeys: changedKeys.length,
  changedByPass: Object.fromEntries(
    DELETED.map((p) => [p, changedKeys.filter((c) => c.fired.includes(p)).length]),
  ),
  unexplainedChangedKeys: unexplained,
  firedKeys: fired.size,
  firedButPdfUnchanged: firedUnchanged.length,
  nondeterministic: nondet,
  outcomeChanges,
  changed: changedKeys,
};
writeFileSync(outFile, JSON.stringify(result, null, 1));
console.log(
  JSON.stringify(
    { ...result, changed: undefined, outcomeChanges: outcomeChanges.map((o) => `${o.test}: ${o.before} -> ${o.after}`) },
    null,
    1,
  ),
);
