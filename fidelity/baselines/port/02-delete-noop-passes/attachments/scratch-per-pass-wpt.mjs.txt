// Scratch (not part of the repo): for every WPT test whose test or reference input changed PDF
// bytes between the before/after runs, report, grouped by the deleted pass(es) that fired on the
// changed node(s): the outcome, each comparison's differing pixels (vs the reference), and the
// test node's ink divergence vs Chrome's test-node ink (fidelity/wpt/lib/pixels.mjs inkDivergence:
// missingFrac of Chrome's significant ink not drawn, bboxShift px).
//   node per-pass-wpt.mjs <before run> <after run> <compare-wpt.json> <out.json>
import { readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { inkDivergence } from "../../wpt/lib/pixels.mjs";

const [beforeDir, afterDir, cmpFile, outFile] = process.argv.slice(2);
const load = (d) => JSON.parse(readFileSync(join(d, "results.json"), "utf8"));
const B = new Map(load(beforeDir).tests.map((t) => [t.test, t]));
const A = new Map(load(afterDir).tests.map((t) => [t.test, t]));
const cmp = JSON.parse(readFileSync(cmpFile, "utf8"));
const changedByUrl = new Map();
for (const c of cmp.changed) {
  const s = changedByUrl.get(c.url) ?? new Set();
  for (const p of c.fired) s.add(p);
  changedByUrl.set(c.url, s);
}
const rows = [];
for (const [name, b] of B) {
  const a = A.get(name);
  const nodes = new Set();
  for (const c of b.comparisons ?? []) {
    nodes.add(c.lhs);
    nodes.add(c.rhs);
  }
  const passes = new Set();
  const changedNodes = [];
  for (const n of nodes) {
    if (changedByUrl.has(n)) {
      changedNodes.push(n === name ? "test" : n);
      for (const p of changedByUrl.get(n)) passes.add(p);
    }
  }
  if (!passes.size) continue;
  const div = (t) =>
    t?.ink && b.chrome?.ink ? inkDivergence(b.chrome.ink, t.ink) : null;
  const px = (t) => (t?.comparisons ?? []).map((c) => c.totalPixels ?? null);
  rows.push({
    test: name,
    passes: [...passes].sort(),
    changedNodes,
    outcome: `${b.outcome} -> ${a.outcome}`,
    chrome: b.chrome?.outcome ?? null,
    diffPixels: { before: px(b), after: px(a) },
    inkVsChrome: { before: div(b), after: div(a) },
  });
}
const groups = {};
for (const r of rows) {
  const g = r.passes.join("+");
  const s = (groups[g] ??= {
    tests: 0,
    outcomes: {},
    diffPixelsSum: { before: 0, after: 0 },
    diffPixels: { better: 0, worse: 0, same: 0 },
    testInkMissingFrac: { better: 0, worse: 0, same: 0, na: 0 },
    testInkBboxShift: { better: 0, worse: 0, same: 0, na: 0 },
  });
  s.tests++;
  s.outcomes[r.outcome] = (s.outcomes[r.outcome] ?? 0) + 1;
  const pb = r.diffPixels.before.reduce((x, y) => x + (y ?? 0), 0);
  const pa = r.diffPixels.after.reduce((x, y) => x + (y ?? 0), 0);
  s.diffPixelsSum.before += pb;
  s.diffPixelsSum.after += pa;
  s.diffPixels[pa < pb ? "better" : pa > pb ? "worse" : "same"]++;
  const ib = r.inkVsChrome.before;
  const ia = r.inkVsChrome.after;
  if (!ib || !ia) {
    s.testInkMissingFrac.na++;
    s.testInkBboxShift.na++;
  } else {
    const cmp3 = (x, y) => (y < x ? "better" : y > x ? "worse" : "same");
    s.testInkMissingFrac[cmp3(ib.missingFrac, ia.missingFrac)]++;
    if (ib.bboxShift == null || ia.bboxShift == null) s.testInkBboxShift.na++;
    else s.testInkBboxShift[cmp3(ib.bboxShift, ia.bboxShift)]++;
  }
}
writeFileSync(outFile, JSON.stringify({ groups, rows }, null, 1));
console.log(JSON.stringify(groups, null, 1));
