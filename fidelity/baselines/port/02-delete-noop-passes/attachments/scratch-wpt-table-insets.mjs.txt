// Scratch (not part of the repo): LOCAL check of the two collapsed_table_medium_border_insets
// passes on WPT inputs. For every td/th the pass moved (pre-deletion dump: layer before vs the
// pass's layer), compare the cell's border box RELATIVE TO ITS TABLE's border box (left/top inset,
// right/bottom inset) in the layer before the pass, after the pass, and in Chrome (same prepared
// HTML, layout-diff's captureSnapshot).
//   node wpt-table-insets.mjs <layout_dump bin> <capture dir> <compare-wpt.json> <out.json>
import { spawnSync } from "node:child_process";
import { readFileSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { captureSnapshot, launchChrome, normalizeSnapshot } from "../../layout-diff/lib/chrome-snapshot.mjs";
import { bundledFontFaceCss } from "../../layout-diff/lib/fonts.mjs";
import { startServer } from "../../wpt/lib/chrome.mjs";

const repoRoot = new URL("../../..", import.meta.url).pathname.replace(/\/$/, "");
const wptRoot = join(repoRoot, "fidelity", ".cache", "wpt");
const CHROME = "/Applications/Google Chrome.app/Contents/MacOS/Google Chrome";
const PASSES = {
  collapsed_table_medium_border_insets: ["top", "bottom"],
  collapsed_table_medium_border_insets_x: ["left", "right"],
};
const TOL = 0.5;
const [bin, capture, cmpFile, outFile] = process.argv.slice(2);
const cmp = JSON.parse(readFileSync(cmpFile, "utf8"));
const fontCss = bundledFontFaceCss(repoRoot);

function layerBoxes(dump, name) {
  const l = dump.layers.find((x) => x.name === name);
  const fields = Array.isArray(l.box_fields) ? l.box_fields : String(l.box_fields).split(",");
  const f = Object.fromEntries(fields.map((k, i) => [k, i]));
  const m = new Map();
  for (const b of l.boxes) {
    if (!m.has(b[f.node_id])) m.set(b[f.node_id], { x: b[f.x], y: b[f.y], w: b[f.w], h: b[f.h] });
  }
  return m;
}
const insets = (cell, table) => ({
  left: cell.x - table.x,
  top: cell.y - table.y,
  right: table.x + table.w - (cell.x + cell.w),
  bottom: table.y + table.h - (cell.y + cell.h),
});

const { server, origin } = await startServer(wptRoot);
const browser = await launchChrome(CHROME);
const rows = [];
try {
  for (const c of cmp.changed.filter((x) => x.fired.some((p) => p in PASSES))) {
    const dir = join(capture, "inputs", c.key);
    const meta = JSON.parse(readFileSync(join(dir, "meta.json"), "utf8"));
    const out = join(tmpdir(), `step02-t-${c.key}.json`);
    const args = ["--html", join(dir, "in.html"), "--out", out];
    for (const f of meta.fonts) args.push("--font", join(capture, "fonts", f));
    for (const [src, h] of Object.entries(meta.images)) args.push("--image", `${src}=${join(capture, "images", h)}`);
    if (meta.baseUrl) args.push("--base-url", meta.baseUrl);
    const r = spawnSync(bin, args, { encoding: "utf8" });
    if (r.status !== 0) throw new Error(r.stderr);
    const dump = JSON.parse(readFileSync(out, "utf8"));
    rmSync(out);
    const base = `${origin}/${c.url.split("/").slice(0, -1).join("/")}/`;
    const html = readFileSync(join(dir, "in.html"), "utf8");
    const inject = `<base href="${base}"><style>@font-face { font-family: Ahem; src: url(/fonts/Ahem.ttf); }</style><style data-vellora-fidelity-fonts>\n${fontCss}\n</style>`;
    const browserHtml = /<head(\s[^>]*)?>/i.test(html)
      ? html.replace(/<head(\s[^>]*)?>/i, (t) => `${t}${inject}`)
      : `<head>${inject}</head>${html}`;
    const { snapshot } = await captureSnapshot(browser, browserHtml, {
      width: dump.viewport.width_px,
      height: dump.viewport.height_px,
    });
    const recs = new Map(normalizeSnapshot(snapshot).filter((x) => x.box).map((x) => [x.path, x]));
    const nodes = new Map(dump.nodes.map((n) => [n.id, n]));
    const names = dump.layers.map((l) => l.name);
    for (const [pass, sides] of Object.entries(PASSES)) {
      const i = names.indexOf(pass);
      const before = layerBoxes(dump, names[i - 1]);
      const after = layerBoxes(dump, pass);
      for (const [id, a] of after) {
        const n = nodes.get(id);
        if (!n || !["td", "th"].includes(n.tag)) continue;
        const b = before.get(id);
        if (!b || (b.x === a.x && b.y === a.y && b.w === a.w && b.h === a.h)) continue;
        let t = n.parent;
        while (t !== null && t !== undefined && nodes.get(t)?.tag !== "table") t = nodes.get(t)?.parent;
        if (t === null || t === undefined) continue;
        const tb = before.get(t);
        const ta = after.get(t);
        const cc = recs.get(n.path);
        const ct = recs.get(nodes.get(t).path);
        const ib = insets(b, tb);
        const ia = insets(a, ta);
        const ic = cc && ct ? insets(cc.box, ct.box) : null;
        for (const side of sides) {
          let verdict = "no-chrome-box";
          if (ic) {
            const e0 = Math.abs(ib[side] - ic[side]);
            const e1 = Math.abs(ia[side] - ic[side]);
            verdict =
              Math.abs(ib[side] - ia[side]) <= 0.01
                ? "side-unchanged"
                : e0 <= TOL && e1 > TOL
                  ? "pass-wrong (Blitz inset == Chrome)"
                  : e1 <= TOL && e0 > TOL
                    ? "pass-right (pass inset == Chrome)"
                    : e1 < e0 - TOL
                      ? "pass-closer"
                      : e1 > e0 + TOL
                        ? "pass-farther"
                        : "same";
          }
          rows.push({
            url: c.url,
            pass,
            cell: n.path,
            side,
            before: ib[side],
            after: ia[side],
            chrome: ic ? ic[side] : null,
            verdict,
          });
        }
      }
    }
    process.stderr.write(".");
  }
} finally {
  await browser.close();
  server.close();
}
const tally = {};
for (const r of rows) {
  tally[r.pass] ??= {};
  tally[r.pass][r.verdict] = (tally[r.pass][r.verdict] ?? 0) + 1;
}
writeFileSync(outFile, JSON.stringify({ tally, rows }, null, 1));
console.log(JSON.stringify({ rows: rows.length, tally }, null, 1));
