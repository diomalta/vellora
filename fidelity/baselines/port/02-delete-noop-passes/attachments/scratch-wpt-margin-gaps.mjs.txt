// Scratch (not part of the repo): LOCAL check of normalize_vertical_margin_collapse on WPT inputs.
// Re-implements the deleted pass's pair rule on the pre-deletion dump's L0 layer to list every
// sibling pair (prev, cur) where it subtracted min(prev.margin_bottom, cur.margin_top) from the
// gap, then compares the gap between the two border boxes in L0 (Blitz, no pass), after the pass
// (L0 gap - amount) and in Chrome (same prepared HTML, layout-diff's captureSnapshot).
//   node wpt-margin-gaps.mjs <layout_dump bin> <capture dir> <compare-wpt.json> <out.json>
import { spawnSync } from "node:child_process";
import { readFileSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { captureSnapshot, launchChrome, normalizeSnapshot } from "../../layout-diff/lib/chrome-snapshot.mjs";
import { bundledFontFaceCss } from "../../layout-diff/lib/fonts.mjs";
import { startServer } from "../../wpt/lib/chrome.mjs";

const repoRoot = new URL("../../..", import.meta.url).pathname.replace(/\/$/, "");
const wptRoot = join(repoRoot, "fidelity", ".cache", "wpt");
const CHROME = "/Applications/Google Chrome.app/Contents/MacOS/Google Chrome";
const COLLAPSIBLE = new Set(
  "address article aside blockquote div footer h1 h2 h3 h4 h5 h6 header main p section".split(" "),
);
const TOL = 0.5;
const [bin, capture, cmpFile, outFile] = process.argv.slice(2);
const cmp = JSON.parse(readFileSync(cmpFile, "utf8"));
const fontCss = bundledFontFaceCss(repoRoot);

function pairsOf(dump) {
  const l0 = dump.layers.find((l) => l.name === "L0");
  const fields = Array.isArray(l0.box_fields) ? l0.box_fields : String(l0.box_fields).split(",");
  const f = Object.fromEntries(fields.map((k, i) => [k, i]));
  const nodes = new Map(dump.nodes.map((n) => [n.id, n]));
  const boxes = l0.boxes.map((b) => ({
    id: b[f.node_id],
    depth: b[f.depth],
    x: b[f.x],
    y: b[f.y],
    w: b[f.w],
    h: b[f.h],
    mt: b[f.margin_top],
    mb: b[f.margin_bottom],
    node: nodes.get(b[f.node_id]),
  }));
  const end = (i) => {
    let e = i + 1;
    while (e < boxes.length && boxes[e].depth > boxes[i].depth) e++;
    return e;
  };
  const out = [];
  const walk = (parent, stop) => {
    const childDepth = boxes[parent].depth + 1;
    let prev = null;
    let i = parent + 1;
    while (i < stop) {
      if (boxes[i].depth !== childDepth) {
        i++;
        continue;
      }
      if (prev !== null) {
        const p = boxes[prev];
        const c = boxes[i];
        const ok = COLLAPSIBLE.has(p.node?.tag) && COLLAPSIBLE.has(c.node?.tag);
        const amount = ok && p.mb > 0 && c.mt > 0 ? Math.min(p.mb, c.mt) : 0;
        if (amount > 0) out.push({ prev: p, cur: c, amount });
      }
      const e = Math.min(end(i), stop);
      walk(i, e);
      prev = i;
      i = e;
    }
  };
  walk(0, boxes.length);
  return out;
}

const { server, origin } = await startServer(wptRoot);
const browser = await launchChrome(CHROME);
const rows = [];
try {
  for (const c of cmp.changed.filter((x) => x.fired.includes("vertical_margin_collapse"))) {
    const dir = join(capture, "inputs", c.key);
    const meta = JSON.parse(readFileSync(join(dir, "meta.json"), "utf8"));
    const out = join(tmpdir(), `step02-g-${c.key}.json`);
    const args = ["--html", join(dir, "in.html"), "--out", out];
    for (const f of meta.fonts) args.push("--font", join(capture, "fonts", f));
    for (const [src, h] of Object.entries(meta.images)) args.push("--image", `${src}=${join(capture, "images", h)}`);
    if (meta.baseUrl) args.push("--base-url", meta.baseUrl);
    const r = spawnSync(bin, args, { encoding: "utf8" });
    if (r.status !== 0) throw new Error(r.stderr);
    const dump = JSON.parse(readFileSync(out, "utf8"));
    rmSync(out);
    const base = `${origin}/${c.url.split("/").slice(0, -1).join("/")}/`;
    const html = readFileSync(join(dir, "in.html"), "utf8");
    const inject = `<base href="${base}"><style>@font-face { font-family: Ahem; src: url(/fonts/Ahem.ttf); }</style><style data-vellora-fidelity-fonts>\n${fontCss}\n</style>`;
    const browserHtml = /<head(\s[^>]*)?>/i.test(html)
      ? html.replace(/<head(\s[^>]*)?>/i, (t) => `${t}${inject}`)
      : `<head>${inject}</head>${html}`;
    const { snapshot } = await captureSnapshot(browser, browserHtml, {
      width: dump.viewport.width_px,
      height: dump.viewport.height_px,
    });
    const recs = new Map(normalizeSnapshot(snapshot).filter((x) => x.box).map((x) => [x.path, x]));
    for (const p of pairsOf(dump)) {
      const cp = recs.get(p.prev.node?.path);
      const cc = recs.get(p.cur.node?.path);
      const l0Gap = p.cur.y - (p.prev.y + p.prev.h);
      const passGap = l0Gap - p.amount;
      const chromeGap = cp && cc ? cc.box.y - (cp.box.y + cp.box.h) : null;
      let verdict = "no-chrome-box";
      if (chromeGap !== null) {
        const e0 = Math.abs(l0Gap - chromeGap);
        const e1 = Math.abs(passGap - chromeGap);
        verdict =
          e0 <= TOL && e1 > TOL
            ? "pass-wrong (Blitz gap == Chrome)"
            : e1 <= TOL && e0 > TOL
              ? "pass-right (pass gap == Chrome)"
              : e1 < e0 - TOL
                ? "pass-closer (neither within 0.5px)"
                : e1 > e0 + TOL
                  ? "pass-farther (neither within 0.5px)"
                  : "same";
      }
      rows.push({
        url: c.url,
        prev: p.prev.node?.path,
        cur: p.cur.node?.path,
        mb: p.prev.mb,
        mt: p.cur.mt,
        amount: p.amount,
        l0Gap,
        passGap,
        chromeGap,
        verdict,
      });
    }
    process.stderr.write(".");
  }
} finally {
  await browser.close();
  server.close();
}
const tally = {};
for (const r of rows) tally[r.verdict] = (tally[r.verdict] ?? 0) + 1;
const urls = {};
for (const r of rows) (urls[r.verdict] ??= new Set()).add(r.url);
writeFileSync(outFile, JSON.stringify({ tally, rows }, null, 1));
console.log(
  JSON.stringify(
    { pairs: rows.length, tally, inputsByVerdict: Object.fromEntries(Object.entries(urls).map(([k, v]) => [k, v.size])) },
    null,
    1,
  ),
);
