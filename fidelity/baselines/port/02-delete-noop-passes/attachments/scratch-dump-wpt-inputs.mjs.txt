// Scratch driver (not part of the repo): run a layout_dump binary over every WPT input captured by
// the recording shim and report, per layout_normalize pass, on how many inputs it changed any box
// or text run (full dump, every box, not only Chrome-matched ones).
//   node dump-wpt-inputs.mjs <layout_dump bin> <capture dir> <out.json> [--jobs 12]
import { spawn } from "node:child_process";
import { mkdirSync, readdirSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { layerEffects } from "/tmp/vellora-step02/layer-identity.mjs";

const [bin, capture, outFile] = process.argv.slice(2);
const jobsIdx = process.argv.indexOf("--jobs");
const jobs = jobsIdx > 0 ? Number(process.argv[jobsIdx + 1]) : 12;
const work = join(tmpdir(), `vellora-step02-dumps-${process.pid}`);
mkdirSync(work, { recursive: true });
const keys = readdirSync(join(capture, "inputs"));

function runOne(key) {
  return new Promise((resolve) => {
    const dir = join(capture, "inputs", key);
    const meta = JSON.parse(readFileSync(join(dir, "meta.json"), "utf8"));
    const out = join(work, `${key}.json`);
    const args = ["--html", join(dir, "in.html"), "--out", out];
    for (const f of meta.fonts) args.push("--font", join(capture, "fonts", f));
    let skipped = null;
    for (const [src, h] of Object.entries(meta.images)) {
      if (src.includes("=")) skipped = `image key with '=' (${src.slice(0, 60)})`;
      args.push("--image", `${src}=${join(capture, "images", h)}`);
    }
    if (meta.baseUrl) args.push("--base-url", meta.baseUrl);
    const child = spawn(bin, args, { stdio: ["ignore", "ignore", "pipe"] });
    let err = "";
    child.stderr.on("data", (d) => {
      err += d;
    });
    child.on("close", (code) => {
      if (code !== 0) {
        resolve({ key, status: "dump-failed", code, err: err.slice(0, 300), skipped });
        return;
      }
      try {
        const dump = JSON.parse(readFileSync(out, "utf8"));
        rmSync(out);
        resolve({ key, status: "ok", effects: layerEffects(dump), skipped });
      } catch (e) {
        resolve({ key, status: "parse-failed", err: String(e).slice(0, 300) });
      }
    });
  });
}

const results = [];
let next = 0;
async function worker() {
  while (next < keys.length) {
    const key = keys[next++];
    results.push(await runOne(key));
    if (results.length % 1000 === 0) process.stderr.write(`${results.length}/${keys.length}\n`);
  }
}
await Promise.all(Array.from({ length: jobs }, worker));
rmSync(work, { recursive: true, force: true });

const summary = { inputs: keys.length, byStatus: {}, passes: {} };
for (const r of results) {
  summary.byStatus[r.status] = (summary.byStatus[r.status] ?? 0) + 1;
  if (r.status !== "ok") continue;
  for (const [name, e] of Object.entries(r.effects)) {
    const s = (summary.passes[name] ??= { inputsChanged: 0, boxesChanged: 0, runsChanged: 0, keys: [] });
    if (e.boxes || e.runs) {
      s.inputsChanged++;
      s.boxesChanged += e.boxes;
      s.runsChanged += e.runs;
      if (s.keys.length < 50) s.keys.push({ key: r.key, boxes: e.boxes, runs: e.runs });
    }
  }
}
writeFileSync(outFile, JSON.stringify({ summary, results }, null, 1));
const brief = { ...summary, passes: Object.fromEntries(Object.entries(summary.passes).map(([k, v]) => [k, { inputsChanged: v.inputsChanged, boxesChanged: v.boxesChanged, runsChanged: v.runsChanged }])) };
console.log(JSON.stringify(brief, null, 1));
