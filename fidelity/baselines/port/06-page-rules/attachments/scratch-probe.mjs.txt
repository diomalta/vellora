// Scratch probe driver (not part of the repo). For every probe in probes.json:
//   - Chrome: the ONE shared reference print path (fidelity/lib/chrome-reference.mjs: print media,
//     bundled Liberation/DejaVu @font-face aliases, page.pdf A4 + preferCSSPageSize).
//   - vellora: renderPdf(html, undefined, { strict: true }) from <repo>/packages/vellora/dist.
// Writes <out>/<name>.chrome.pdf and <out>/<name>.vellora.pdf (or <name>.vellora.error.json).
//   node probe.mjs <repo> <probes.json> <outDir> [--only a,b] [--no-chrome]
import { mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { pathToFileURL } from "node:url";

const [repo, probesFile, outDir, ...rest] = process.argv.slice(2);
const onlyIdx = rest.indexOf("--only");
const only = onlyIdx >= 0 ? new Set(rest[onlyIdx + 1].split(",")) : null;
const noChrome = rest.includes("--no-chrome");
const ref = await import(pathToFileURL(join(repo, "fidelity/lib/chrome-reference.mjs")).href);
const vellora = await import(pathToFileURL(join(repo, "packages/vellora/dist/index.js")).href);
const probes = JSON.parse(readFileSync(probesFile, "utf8"));
mkdirSync(outDir, { recursive: true });
const chrome = noChrome ? null : await ref.launchReferenceChrome();
const meta = { chrome: chrome?.version ?? null, printOptions: ref.PRINT_OPTIONS, probes: [] };
try {
  for (const [name, html] of Object.entries(probes)) {
    if (only && !only.has(name)) continue;
    if (chrome) {
      const pdf = await ref.printPdf(chrome, ref.referenceHtml(html));
      writeFileSync(join(outDir, `${name}.chrome.pdf`), pdf);
    }
    try {
      const pdf = await vellora.renderPdf(html, undefined, { strict: true });
      writeFileSync(join(outDir, `${name}.vellora.pdf`), pdf);
    } catch (e) {
      writeFileSync(
        join(outDir, `${name}.vellora.error.json`),
        JSON.stringify({ name: e.name, message: e.message, diagnostics: e.diagnostics ?? null }),
      );
    }
    meta.probes.push(name);
  }
} finally {
  await chrome?.close();
}
writeFileSync(join(outDir, "meta.json"), JSON.stringify(meta, null, 1));
console.log(JSON.stringify(meta));
