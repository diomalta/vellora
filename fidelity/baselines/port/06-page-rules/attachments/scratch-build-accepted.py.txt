"""Build accepted.json (13 entries) and accepted-evidence.json for port step 06.

usage: build_accepted.py <run dir> <step05 run dir> <evidence dir> <accepted.json> <accepted-evidence.json>
"""
import json
import re
import subprocess
import sys
import unicodedata

run, old_run, E, out_accept, out_evidence = sys.argv[1:6]
EVIDENCE_COMMON = (
    "Chrome 155.0.8059.85 local macOS (INDICATIVE), 2026-10-21; old = port/05 run 20261021T142201Z "
    "(e9dc7e4), new = port/06 run of c9fe9f5; numbers: attachments/accepted-evidence.json; label "
    "legend: baselines/port/README.md (06-page-rules)"
)


def keyed(path):
    d = json.load(open(path))
    seen, out = {}, {}
    for p in d["pairs"]:
        norm = unicodedata.normalize("NFC", p["text"])
        seen[norm] = seen.get(norm, 0) + 1
        out[f"{norm}#{seen[norm]}"] = p
    return out


probe_rel = "layout-diff/text-stress-ptbr/run-all"
before = keyed(f"{old_run}/{probe_rel}/pdf-probe.json")
after = keyed(f"{run}/{probe_rel}/pdf-probe.json")

WORD = re.compile(
    r'<page\b[^>]*>|<word xMin="([\d.]+)" yMin="([\d.]+)" xMax="([\d.]+)" yMax="([\d.]+)">([^<]*)</word>'
)


def footer_de(pdf):
    """Per page, the (x, yMin) of the bottom-band "de" (yMin > 790 pt), pdftotext -bbox."""
    out = subprocess.run(
        ["/opt/homebrew/bin/pdftotext", "-bbox", pdf, "-"], capture_output=True, text=True
    ).stdout
    page, rows = 0, {}
    for m in WORD.finditer(out):
        if m.group(0).startswith("<page"):
            page += 1
        elif float(m.group(2)) > 790 and m.group(5) == "de":
            rows[page] = (round(float(m.group(1)), 2), round(float(m.group(2)), 2))
    return rows


chrome_footer = footer_de(f"{run}/{probe_rel}/chrome.pdf")
new_footer = footer_de(f"{run}/{probe_rel}/vellora.pdf")
old_footer = footer_de(f"{old_run}/{probe_rel}/vellora.pdf")

accept, evidence = [], {"pdfProbePairing": [], "wpt": {}}
for k in ["de#33", "de#61", "de#87", "de#115", "de#155", "de#168"]:
    a, b = after[k], before[k]
    pg = a["vellora"]["page"]
    cf, nf, of = chrome_footer.get(pg), new_footer.get(pg), old_footer.get(pg)
    row = {
        "key": k,
        "vellora": {"before": b["vellora"], "after": a["vellora"]},
        "pairedChromeWord": a["chrome"],
        "dx": {"before": b["dx_pt"], "after": a["dx_pt"]},
        "dy": {"before": b["dy_pt"], "after": a["dy_pt"]},
        "byPosition": {
            "page": pg,
            "chromeFooterDe": cf,
            "velloraFooterDeBefore": of,
            "velloraFooterDeAfter": nf,
            "absDxBefore": round(abs(of[0] - cf[0]), 2),
            "absDxAfter": round(abs(nf[0] - cf[0]), 2),
            "absDyBefore": round(abs(of[1] - cf[1]), 2),
            "absDyAfter": round(abs(nf[1] - cf[1]), 2),
        },
    }
    evidence["pdfProbePairing"].append(row)
    for axis in ("dx", "dy"):
        accept.append(
            {
                "key": f"pdf-probe|text-stress-ptbr|{k}|{axis}",
                "after": abs(a[f"{axis}_pt"]),
                "label": "pdf-probe-pairing-artefact",
                "reason": (
                    f"Fits GATE-A1 (oracle artefact): the k-th-occurrence pair is two different word "
                    f"instances. vellora's {k} is the page-{pg} FOOTER 'de' ({a['vellora']['x']:.2f}, "
                    f"{a['vellora']['y']:.2f}) pt; the Chrome word it is paired with is a BODY 'de' at "
                    f"({a['chrome']['x']:.2f}, {a['chrome']['y']:.2f}) pt (vellora 8 pages vs Chrome 7, "
                    f"and the extractor emits the footers at different occurrence positions). The "
                    f"{axis} change ({b[f'{axis}_pt']} -> {a[f'{axis}_pt']} pt) is the footer moving to "
                    f"where Chrome draws it: by position, page-{pg} footer 'de' vs Chrome's ({cf[0]}, "
                    f"{cf[1]}) pt went |dx| {row['byPosition']['absDxBefore']} -> "
                    f"{row['byPosition']['absDxAfter']} pt, |dy(yMin)| "
                    f"{row['byPosition']['absDyBefore']} -> {row['byPosition']['absDyAfter']} pt "
                    f"(yMin carries each producer's declared ascent, step 04; PyMuPDF baseline of the "
                    f"8pt Liberation Sans footer: Chrome 819.75, vellora 819.41 pt)."
                ),
                "evidence": f"pdftotext 26.06.0 -bbox of layout-diff text-stress-ptbr chrome.pdf / vellora.pdf of both runs; {EVIDENCE_COMMON}",
            }
        )

pages = {
    "chrome": json.load(open(f"{E}/chrome-wpt-pagesize/pagesizes.json")),
    "vellora": json.load(open(f"{E}/vellora-wpt-pagesize/pagesizes.json")),
}
evidence["wpt"]["css/css-page/page-size-015-print.tentative.html"] = {
    "chromePageSizes": {
        k: v for k, v in pages["chrome"]["files"].items() if "page-size-015" in k
    },
    "velloraPageSizes": {k: v for k, v in pages["vellora"].items() if "page-size-015" in k},
}
evidence["wpt"]["css/css-page/page-name-001-print.html (reported, flagged)"] = {
    "chromePageSizes": {k: v for k, v in pages["chrome"]["files"].items() if "page-name-001" in k},
    "velloraPageSizes": {k: v for k, v in pages["vellora"].items() if "page-name-001" in k},
}
accept.append(
    {
        "key": "wpt|css/css-page/page-size-015-print.tentative.html",
        "after": "fail",
        "label": "wpt-false-pass",
        "reason": (
            "pass -> fail. Fits GATE-A1 (FALSE PASS): the old test equalled its reference only "
            "because both lost their page size. Chrome prints test AND reference at 540 x 432 pt "
            "(7.5in x 6in, the tentative @page width/height descriptors). Old vellora printed both "
            "at 360 x 216 pt (the 5in x 3in UA box): its @page reader did not strip the CSS comment "
            "in front of the test's `size: 1234px`, so it read the property name '/* ... */ size' "
            "and dropped the declaration; the reference has no size. Now comments are stripped and "
            "the test gets `size: 1234px` (925.5 x 925.5 pt, CSS Paged Media 3), the reference "
            "keeps 360 x 216; vellora does not implement the tentative width/height descriptors "
            "(follow-up)."
        ),
        "evidence": f"pdfinfo page sizes: Chrome with the WPT harness print geometry, vellora old = step-05 addon sha256 0211dcfb... with the step-05 injection, new = c9fe9f5; {EVIDENCE_COMMON}",
    }
)
json.dump({"acceptedRegressions": accept}, open(out_accept, "w"), ensure_ascii=False, indent=1)
json.dump(evidence, open(out_evidence, "w"), ensure_ascii=False, indent=1)
print(len(accept), "acceptances")
