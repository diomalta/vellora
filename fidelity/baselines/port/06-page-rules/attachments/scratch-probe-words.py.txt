"""Print every word (page, text, x, baseline, size, font) of <out>/<name>.{chrome,vellora}.pdf.

usage: probe_words.py <outDir> [names...]   (run with fidelity/.venv python -I)
"""
import glob
import json
import os
import sys

import pymupdf

out = sys.argv[1]
names = sys.argv[2:] or sorted(
    {os.path.basename(p).split(".")[0] for p in glob.glob(os.path.join(out, "*.pdf"))}
)


def words(path):
    res = []
    with pymupdf.open(path) as doc:
        for pi, page in enumerate(doc, start=1):
            raw = page.get_text("rawdict")
            for b in raw["blocks"]:
                if b.get("type") != 0:
                    continue
                for line in b["lines"]:
                    for span in line["spans"]:
                        cur = None
                        for ch in span["chars"]:
                            if ch["c"].isspace():
                                if cur:
                                    res.append(cur)
                                cur = None
                                continue
                            if cur is None:
                                cur = {
                                    "page": pi,
                                    "text": "",
                                    "x": round(ch["origin"][0], 2),
                                    "baseline": round(ch["origin"][1], 2),
                                    "size": round(span["size"], 2),
                                    "font": span["font"],
                                    "color": "#%06x" % span["color"],
                                }
                            cur["text"] += ch["c"]
                            cur["x1"] = round(ch["bbox"][2], 2)
                        if cur:
                            res.append(cur)
            res.append({"page": pi, "pageSize": [round(page.rect.width, 2), round(page.rect.height, 2)]})
    return res


result = {}
for n in names:
    result[n] = {}
    for who in ("chrome", "vellora"):
        p = os.path.join(out, f"{n}.{who}.pdf")
        if os.path.exists(p):
            result[n][who] = words(p)
        else:
            e = os.path.join(out, f"{n}.{who}.error.json")
            result[n][who] = json.load(open(e)) if os.path.exists(e) else None
print(json.dumps(result, ensure_ascii=False, indent=0))
