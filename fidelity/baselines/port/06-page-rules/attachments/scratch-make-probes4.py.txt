"""probes4.json: page progression of an rtl root (page 1 is a left page per css-page-3)."""
import json
import sys

out = sys.argv[1]
THREE = (
    '<div style="height: 50px">pagina um</div>'
    '<div style="break-before: page; height: 50px">pagina dois</div>'
    '<div style="break-before: page; height: 50px">pagina tres</div>'
)


def doc(css, body, html_attrs=""):
    return (
        f'<!DOCTYPE html><html{html_attrs}><head><meta charset="utf-8"><style>' + css
        + ' body { margin: 0; font-family: "Liberation Serif"; font-size: 11pt; }'
        + "</style></head><body>" + body + "</body></html>"
    )


probes = {
    "rtl_left_right_vertical": doc(
        "@page { size: A4; margin: 10mm; } @page :left { margin-top: 40mm; }"
        " @page :right { margin-top: 20mm; }",
        THREE,
        ' dir="rtl"',
    ),
}
json.dump(probes, open(out, "w"), ensure_ascii=False, indent=1)
print(len(probes), "probes")
