"""Census of @page usage in the WPT reftests vellora runs (test + every reference file).

usage: page_rules_census.py <results.json> <wpt root> [out.json]
Classifies each file by: number of @page rules (in <style> blocks AND linked .css inlined? no:
raw file only, plus linked stylesheets resolved relative to the file), selectors used, margin boxes.
"""
import json
import os
import re
import sys
from collections import Counter

results_path, wpt_root = sys.argv[1], sys.argv[2]
out_path = sys.argv[3] if len(sys.argv) > 3 else None
d = json.load(open(results_path))

STYLE_RE = re.compile(r"<style[^>]*>(.*?)</style", re.S | re.I)
LINK_RE = re.compile(r"<link[^>]+>", re.I)
HREF_RE = re.compile(r"href\s*=\s*[\"']([^\"']+)[\"']", re.I)
COMMENT_RE = re.compile(r"/\*.*?\*/", re.S)


def css_of(rel):
    p = os.path.join(wpt_root, rel)
    try:
        html = open(p, encoding="utf-8", errors="replace").read()
    except OSError:
        return None
    css = []
    for m in STYLE_RE.finditer(html):
        css.append(m.group(1))
    for m in LINK_RE.finditer(html):
        tag = m.group(0)
        if "stylesheet" not in tag.lower():
            continue
        h = HREF_RE.search(tag)
        if not h:
            continue
        href = h.group(1)
        if href.startswith("/"):
            cp = os.path.join(wpt_root, href.lstrip("/"))
        else:
            cp = os.path.normpath(os.path.join(os.path.dirname(p), href))
        try:
            css.append(open(cp, encoding="utf-8", errors="replace").read())
        except OSError:
            pass
    return COMMENT_RE.sub("", "\n".join(css))


PAGE_RE = re.compile(r"@page\b([^{]*)\{", re.I)


def page_rules(css):
    rules = []
    for m in PAGE_RE.finditer(css):
        sel = m.group(1).strip()
        rules.append(sel)
    return rules


def classify(rel):
    css = css_of(rel)
    if css is None:
        return None
    rules = page_rules(css)
    margin_boxes = sorted(set(re.findall(r"@((?:top|bottom|left|right)-[a-z-]+)", css, re.I)))
    return {"rules": rules, "marginBoxes": margin_boxes}


rows = []
for t in d["tests"]:
    files = {t["test"]}
    for c in t.get("comparisons") or []:
        files.add(c["lhs"])
        files.add(c["rhs"])
    info = {f: classify(f) for f in sorted(files)}
    n = max((len(v["rules"]) for v in info.values() if v), default=0)
    sels = sorted({s for v in info.values() if v for s in v["rules"]})
    mbs = sorted({b for v in info.values() if v for b in v["marginBoxes"]})
    if n == 0 and not mbs:
        continue
    rows.append(
        {
            "test": t["test"],
            "mode": t.get("mode"),
            "outcome": t["outcome"],
            "chrome": (t.get("chrome") or {}).get("outcome"),
            "cause": (t.get("cause") or {}).get("feature"),
            "maxRulesPerFile": n,
            "selectors": sels,
            "marginBoxes": mbs,
            "perFile": {f: v for f, v in info.items() if v and (v["rules"] or v["marginBoxes"])},
        }
    )

print("tests with @page anywhere:", len(rows))
print("by mode:", Counter(r["mode"] for r in rows))
print("multi-rule (some file has >1 @page):", sum(1 for r in rows if r["maxRulesPerFile"] > 1))
sel_counter = Counter()
for r in rows:
    for s in r["selectors"]:
        sel_counter[s or "<none>"] += 1
print("selectors:", sel_counter.most_common(40))
print("outcome x chrome (multi or selector):")
interesting = [r for r in rows if r["maxRulesPerFile"] > 1 or any(r["selectors"]) or r["marginBoxes"]]
print(Counter((r["outcome"], r["chrome"]) for r in interesting))
print("screen tests with @page:", [(r["test"], r["outcome"], r["selectors"]) for r in rows if r["mode"] == "screen"][:40])
if out_path:
    json.dump(rows, open(out_path, "w"), indent=1)
