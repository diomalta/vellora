// Page sizes vellora gives WPT print-reftest files, old vs new harness + build:
//   old: step-05 build (addon sha256 0211dcfb…, = run 20261021T142201Z) with the step-05 harness
//        injection (UA print box merged at the start of the first @page block);
//   new: this build with this harness (fidelity/wpt/lib/prepare.mjs prepareNode).
//   node vellora_wpt_pagesize.mjs <repo> <oldRepo> <wptRoot> <outDir> <rel> [<rel> ...]
import { execFileSync } from "node:child_process";
import { mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { pathToFileURL } from "node:url";

const [repo, oldRepo, wptRoot, out, ...rels] = process.argv.slice(2);
mkdirSync(out, { recursive: true });
const prep = await import(pathToFileURL(join(repo, "fidelity/wpt/lib/prepare.mjs")).href);
const neu = await import(pathToFileURL(join(repo, "packages/vellora/dist/index.js")).href);
const old = await import(pathToFileURL(join(oldRepo, "packages/vellora/dist/index.js")).href);
const sizes = (file) =>
  execFileSync("pdfinfo", ["-f", "1", "-l", "9", file], { encoding: "utf8" })
    .split("\n")
    .filter((l) => /^Page\s+\d+ size|^Pages:/.test(l))
    .map((l) => l.trim());
const result = {};
for (const rel of rels) {
  const content = readFileSync(join(wptRoot, rel), "utf8");
  const url = `http://web-platform.test:8000/${rel}`;
  const prepared = prep.prepareNode({ wptRoot, rel, url, content, mode: "print" });
  const newPdf = join(out, `${rel.replaceAll("/", "_")}.vellora-new.pdf`);
  writeFileSync(
    newPdf,
    await neu.renderPdf(prepared.html, undefined, {
      strict: true,
      fonts: prepared.fonts,
      images: prepared.images,
      baseUrl: prepared.baseUrl,
    }),
  );
  // Step-05 harness: merge the UA box at the start of the first @page block.
  const at = prep.firstAtPageBodyOffset(content);
  const oldHtml =
    at === -1
      ? content
      : `${content.slice(0, at)} ${prep.PRINT_PAGE_CSS}; ${content.slice(at)}`;
  const oldPdf = join(out, `${rel.replaceAll("/", "_")}.vellora-old.pdf`);
  writeFileSync(
    oldPdf,
    await old.renderPdf(oldHtml, undefined, {
      strict: true,
      fonts: prepared.fonts,
      images: prepared.images,
      baseUrl: prepared.baseUrl,
    }),
  );
  result[rel] = { old: sizes(oldPdf), new: sizes(newPdf), oldInjection: at === -1 ? "none" : "merged" };
}
console.log(JSON.stringify(result, null, 1));
