// Scratch (not part of the repo): assemble chrome-probes-table-passes.json from the Chrome probe
// outputs and vellora layout_dump probes (step-02 build = before, step-03 build = after).
//   node build-probe-attachment.mjs <out.json>
import { spawnSync } from "node:child_process";
import { createHash } from "node:crypto";
import { mkdirSync, readFileSync, writeFileSync } from "node:fs";

const S = "/tmp/vellora-step03";
const REPO = "<repo>";
const BEFORE = `${process.env.HOME}/.cache/vellora-fidelity/step02-delete-noop-passes-evidence-20261021/layout_dump-after`;
const AFTER = `${REPO}/target/release/examples/layout_dump`;
const sets = [
  { probes: `${S}/probes.json`, chrome: `${S}/chrome-probes.json`, viewport: 657 },
  { probes: `${S}/probes2.json`, chrome: `${S}/chrome-probes2-657.json`, viewport: 657, only: ["fixed_colspans"] },
  { probes: `${S}/probes2.json`, chrome: `${S}/chrome-probes2-672.json`, viewport: 672, only: ["boleto_rowspan_grid"] },
  { probes: `${S}/probes3.json`, chrome: `${S}/chrome-probes3.json`, viewport: 672 },
  { probes: `${S}/probes4.json`, chrome: `${S}/chrome-probes4.json`, viewport: 657 },
  { probes: `${S}/probes-invoice.json`, chrome: `${S}/chrome-probes-invoice.json`, viewport: 672, htmlFile: "crates/vellora-core/tests/fixtures/invoice.html" },
];
const r3 = (v) => Math.round(v * 1000) / 1000;
function velloraBoxes(bin, name, html) {
  const dir = `${S}/attach-work`;
  mkdirSync(dir, { recursive: true });
  const h = `${dir}/${name}.html`;
  const d = `${dir}/${name}.${bin === BEFORE ? "before" : "after"}.dump.json`;
  writeFileSync(h, html);
  const r = spawnSync(bin, ["--html", h, "--out", d], { encoding: "utf8" });
  if (r.status !== 0) throw new Error(r.stderr);
  const dump = JSON.parse(readFileSync(d, "utf8"));
  const nodes = new Map(dump.nodes.map((n) => [n.id, n]));
  const pick = (layer) => {
    const f = Object.fromEntries(String(layer.box_fields).split(",").map((k, i) => [k, i]));
    const rf = Object.fromEntries(String(layer.run_fields).split(",").map((k, i) => [k, i]));
    const out = {};
    let n = 0;
    layer.boxes.forEach((b, bi) => {
      const node = nodes.get(b[f.node_id]);
      if (!node || node.kind !== "element") return;
      if (!(node.id_attr || ["table", "td", "th"].includes(node.tag))) return;
      const key = node.id_attr || `${node.tag}#${n++}`;
      out[key] = {
        x: r3(b[f.x]),
        w: r3(b[f.w]),
        runs: layer.runs
          .filter((q) => q[rf.box_index] === bi)
          .map((q) => [r3(q[rf.origin_x]), r3(q[rf.origin_x] + q[rf.advance])]),
      };
    });
    return out;
  };
  return {
    layers: dump.layers.map((l) => l.name),
    L0: pick(dump.layers[0]),
    final: pick(dump.layers.at(-1)),
  };
}
const out = {
  what:
    "Chrome border boxes (getBoundingClientRect) and text line rects (Range.getClientRects per text node) of every element with an id and every table/td/th of the probe documents behind the port-step-03 tests in crates/vellora-core/tests/layout_fidelity.rs, laid out by fidelity/lib/chrome-reference.mjs (print media, bundled Liberation @font-face block) at the viewport vellora hands Blitz (A4 content width: 657 px at 18 mm margins, 672 px at 16 mm), next to vellora's boxes and text runs (layout_dump: L0 = raw Blitz walk, final = last layout_normalize layer) with the step-02 build (before) and this step's build (after). CSS px, content-box origin.",
  env: {
    date: "2026-10-21",
    os: "darwin 27.0.0 arm64 (macOS 27.0.1) / Apple M5 Pro",
    node: process.version,
    chrome: "Google Chrome 155.0.8059.85 (local, INDICATIVE)",
    blitz: "blitz-dom 0.3.0-beta.4 (crates.io)",
    before: "phaseb/02-delete-noop-passes code 6e5cb8d (layout_dump from ~/.cache/vellora-fidelity/step02-delete-noop-passes-evidence-20261021/layout_dump-after)",
    after: "phaseb/03-table-passes code commit (layout_dump built from it)",
    drivers: "attachments/scratch-chrome-probe.mjs.txt, scratch-vellora-probe.mjs.txt, scratch-build-probe-attachment.mjs.txt",
    note: "Generic `sans-serif` resolves to system Helvetica in Chrome (generic families do not consult @font-face), so text-bearing probes name \"Liberation Sans\"; fixed_colspans and boleto_rowspan_grid keep the tests' sans-serif because their geometry does not depend on text.",
  },
  probes: {},
};
for (const set of sets) {
  const probes = JSON.parse(readFileSync(set.probes, "utf8"));
  const chrome = JSON.parse(readFileSync(set.chrome, "utf8"));
  for (const [name, html] of Object.entries(probes)) {
    if (set.only && !set.only.includes(name)) continue;
    const c = {};
    for (const [k, v] of Object.entries(chrome.probes[name])) {
      if (set.htmlFile && !/^t[dh]#([7-9]|1[0-4])$/.test(k)) continue;
      c[k] = {
        x: r3(v.x),
        w: r3(v.w),
        textAlign: v.textAlign,
        lines: v.lines.map((l) => [l.text, r3(l.x), r3(l.right)]),
      };
    }
    const entry = {
      viewport: set.viewport,
      html: set.htmlFile
        ? { file: set.htmlFile, sha256: createHash("sha256").update(html).digest("hex") }
        : html,
      chrome: c,
      vellora_before: velloraBoxes(BEFORE, name, html),
      vellora_after: velloraBoxes(AFTER, name, html),
    };
    if (set.htmlFile) {
      for (const side of ["vellora_before", "vellora_after"]) {
        for (const layer of ["L0", "final"]) {
          entry[side][layer] = Object.fromEntries(
            Object.entries(entry[side][layer]).filter(([k]) => /^t[dh]#([7-9]|1[0-4])$/.test(k)),
          );
        }
      }
    }
    out.probes[name] = entry;
  }
}
writeFileSync(process.argv[2], `${JSON.stringify(out, null, 1)}\n`);
console.log("probes:", Object.keys(out.probes).join(", "));
