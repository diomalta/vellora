// Scratch (not part of the repo): ablation summary of port step 03's variants, each gated against
// port/02-delete-noop-passes by a full `run-all --gate` run (same env, 2026-10-21).
//   node build-ablation.mjs <out.json>
import { readFileSync, writeFileSync } from "node:fs";

const C = `${process.env.HOME}/.cache/vellora-fidelity`;
const E = `${C}/step03-table-passes-evidence-20261021`;
const base02 = `${C}/run-all-02-delete-noop-passes-20261021T102146Z`;
const variants = [
  { id: "V1-align", what: "alignment-aware text move; fixed_table_widths + compact kept", run: `${E}/v1-align/run` },
  { id: "V3-nofixed", what: "V1 + fixed_table_widths deleted (compact kept)", run: `${E}/v3-nofixed/run` },
  { id: "V4-nofixed-nocompact", what: "V3 + auto_table_compact_columns deleted", run: `${E}/v4-nofixed-nocompact/run` },
  { id: "final", what: "V3 + table_percent_widths skips table-layout: fixed tables (committed)", run: `${C}/run-all-03-table-passes-20261021T114420Z` },
];
const KEYS = [
  "invoice|REF-W|oracles:layout|plain|criterion#1|maxMm",
  "invoice|REF-W|oracles:layout|plain|criterion#1|p95Mm",
  "danfe|REF-W|oracles:layout|plain|criterion#1|maxMm",
  "boleto-febraban|REF-W|oracles:layout|plain|criterion#1|maxMm",
  "text-stress-ptbr|REF-W|oracles:layout|plain|criterion#1|maxMm",
  "boleto-febraban|BOL-X1|corpus||criterion#1|found",
  "danfe|DANFE-X1|corpus||criterion#1|found",
  "danfe|DANFE-F11|corpus||criterion#1|glyphsAttributed",
  "danfe|DANFE-B5|corpus||criterion#1|excess",
];
const wptSignal = (results) => {
  const r = JSON.parse(readFileSync(results, "utf8"));
  let px = 0;
  let pass = 0;
  for (const t of r.tests) {
    if (!/^css\/(CSS2\/tables|css-tables)\//.test(t.test)) continue;
    px += t.comparisons?.[0]?.totalPixels ?? 0;
    if (t.outcome === "pass") pass++;
  }
  return { tablesSuitesPass: pass, tablesSuitesDiffPixels: px };
};
const out = {
  what: "Ablation of the table passes (port step 03). Every variant is a full run-all --gate run against fidelity/baselines/port/02-delete-noop-passes, same environment (darwin 27.0.0 arm64 / Apple M5 Pro, node v22.23.2, rustc 1.96.0, Chrome 155.0.8059.85 local INDICATIVE, Chrome WPT data reused from the Phase A beta.4 run). tablesSuitesDiffPixels = sum over css/CSS2/tables + css/css-tables of the differing-pixel count of each test's first comparison (0 for passes): a continuous signal under the binary WPT outcome.",
  baseline02: { ...wptSignal(`${base02}/wpt/results.json`) },
  variants: [],
};
for (const v of variants) {
  const g = JSON.parse(readFileSync(`${v.run}/gate/gate-report.json`, "utf8"));
  const raw = (() => {
    try {
      return JSON.parse(readFileSync(`${v.run}/gate-without-acceptances/gate-report.json`, "utf8"));
    } catch {
      return g;
    }
  })();
  const m = new Map(raw.items.map((i) => [i.key, i]));
  const wpt = JSON.parse(readFileSync(`${v.run}/wpt/outcomes.json`, "utf8"));
  let pass = 0;
  let clean = 0;
  for (const [k, o] of Object.entries(wpt.results)) {
    if (o !== "pass") continue;
    pass++;
    if (!wpt.suspectPasses?.[k]) clean++;
  }
  const byGroup = {};
  for (const i of raw.items) {
    const grp = i.group ?? i.check;
    byGroup[grp] ??= {};
    byGroup[grp][i.class] = (byGroup[grp][i.class] ?? 0) + 1;
  }
  out.variants.push({
    id: v.id,
    what: v.what,
    gateVs02WithoutAcceptances: { verdict: raw.verdict, counts: raw.counts },
    byGroup,
    metrics: Object.fromEntries(KEYS.map((k) => [k, m.get(k) ? { before: m.get(k).before, after: m.get(k).after, class: m.get(k).class } : "unchanged"])),
    wpt: { pass, clean, total: Object.keys(wpt.results).length, ...wptSignal(`${v.run}/wpt/results.json`) },
  });
}
writeFileSync(process.argv[2], `${JSON.stringify(out, null, 1)}\n`);
for (const v of out.variants) console.log(v.id, JSON.stringify(v.gateVs02WithoutAcceptances), JSON.stringify(v.wpt));
console.log("02", JSON.stringify(out.baseline02));
