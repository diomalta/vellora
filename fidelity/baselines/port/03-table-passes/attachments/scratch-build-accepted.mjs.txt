// Scratch (not part of the repo): build accepted.json for port step 03 from the run's gate report.
//   node build-accepted.mjs <gate-report.json> <old run dir> <new run dir> <out accepted.json> <out evidence.json>
// Every gating item must be a pdf-probe word; each gets one of two labels:
//   pdf-probe-pairing-artefact          the gate pairs the k-th occurrence of a token on each side;
//                                       by position (nearest same-text Chrome word on the same page,
//                                       or in the same column when the pages hold different rows)
//                                       the word got closer to Chrome or moved by < 1 pt
//   coincidental-match-of-wrong-geometry the word's old position came from fixed_table_widths' equal
//                                       tracks (which ignore <col> widths); the new one is raw Blitz
//                                       (L0 == final layer) whose residual is the collapsed-border
//                                       gutter of Blitz's fixed tables (separate defect, follow-up)
import { readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";

const [gateFile, oldDir, newDir, outAccept, outEvidence] = process.argv.slice(2);
const gate = JSON.parse(readFileSync(gateFile, "utf8"));
const load = (dir, fx, f) => JSON.parse(readFileSync(join(dir, "layout-diff", fx, "run-all", f), "utf8"));
const cache = new Map();
const get = (dir, fx, f) => {
  const k = `${dir}|${fx}|${f}`;
  if (!cache.has(k)) cache.set(k, load(dir, fx, f));
  return cache.get(k);
};
const r2 = (v) => (v == null ? null : Math.round(v * 100) / 100);

function occurrence(p, text, occ) {
  return p.pairs.filter((x) => x.text === text)[occ - 1] ?? null;
}
/** Nearest Chrome word with the same text: same page first, else same column on any page. */
function nearest(p, pair, text) {
  const cands = p.pairs.filter((x) => x.chrome && x.text.toLowerCase() === text.toLowerCase()).map((x) => x.chrome);
  let best = null;
  for (const c of cands) {
    if (c.page !== pair.vellora.page) continue;
    const d = Math.hypot(pair.vellora.x - c.x, pair.vellora.y - c.y);
    if (!best || d < best.d) best = { d, dx: pair.vellora.x - c.x, dy: pair.vellora.y - c.y, page: c.page, how: "same page" };
  }
  return best;
}
function sameColumn(p, pair, text) {
  // closest Chrome x among same-text words anywhere (column alignment), when pages hold other rows
  let best = null;
  for (const x of p.pairs) {
    if (!x.chrome || x.text !== text) continue;
    const dx = pair.vellora.x - x.chrome.x;
    if (!best || Math.abs(dx) < Math.abs(best.dx)) best = { dx, page: x.chrome.page, how: "same column (any page)" };
  }
  return best;
}

const DANFE_TABLE_NOTE =
  "fixed_table_widths split every table-layout: fixed table into equal tracks and ignored the <col> widths (CSS 2.2 §17.5.2.1) that Chrome 155 and Blitz 0.3.0-beta.4 honour (attachments/chrome-probes-table-passes.json fixed_cols: Chrome 60/220/120, Blitz 60/220/120, old pass 133.33 x 3). The DANFE grids are built from <col> widths, so the old cell edges were wrong by construction and agreed with Chrome near these words only where an equal-track edge happened to fall near a <col> edge (table 5: <col> widths 6.86 / 6.86 / 6.78 cm, equal track 6.83 cm), while across the DANFE the same pass put text up to tens of pt off (1465 pdf-probe words improve in this step). The new position is raw Blitz: the layout-diff L0 dx equals the final-layer dx for every one of these runs (no pass moves them). Its residual is Blitz's collapsed-border model: it lays each collapsed border out as a whole-width gutter between <col>-length tracks, so cells drift right by about one border width (0.5pt = 0.67 px) per column and the last cells overflow the table (DANFE table 7: Chrome td x 0.5 / 466.16 / 667.59, Blitz +0.50 / +4.84 / +6.41 px). Not a false pass or oracle artefact in the literal sense of the merge rule: the old value was closer to Chrome, but only by coincidence of a wrong mechanism; flagged for the orchestrator. Follow-up: Blitz collapsed-border gutters in fixed tables.";

const accepted = [];
const evidence = [];
for (const it of gate.items.filter((x) => x.gating)) {
  if (!it.key.startsWith("pdf-probe|")) throw new Error(`unexpected gating item ${it.key}`);
  const [, fx, tok, coord] = it.key.split("|");
  const i = tok.lastIndexOf("#");
  const text = tok.slice(0, i);
  const occ = Number(tok.slice(i + 1));
  const pNew = get(newDir, fx, "pdf-probe.json");
  const pOld = get(oldDir, fx, "pdf-probe.json");
  const nNew = occurrence(pNew, text, occ);
  const nOld = occurrence(pOld, text, occ);
  const nearNew = nearest(pNew, nNew, text);
  const nearOld = nOld ? nearest(pOld, nOld, text) : null;
  const colNew = sameColumn(pNew, nNew, text);
  const colOld = nOld ? sameColumn(pOld, nOld, text) : null;
  const k = coord === "dx" ? "dx" : "dy";
  const posNew = nearNew ? Math.abs(nearNew[k]) : null;
  const posOld = nearOld ? Math.abs(nearOld[k]) : null;
  const pagesDiffer = pNew.pages.vellora !== pNew.pages.chrome;
  const e = {
    key: it.key,
    before: it.before,
    after: it.after,
    occurrencePair: {
      old: nOld && { vellora: nOld.vellora, chrome: nOld.chrome },
      new: { vellora: nNew.vellora, chrome: nNew.chrome },
    },
    nearestSameTextSamePage: { old: nearOld && r2(nearOld[k]), new: nearNew && r2(nearNew[k]) },
    nearestSameTextSameColumn: coord === "dx" ? { old: colOld && r2(colOld.dx), new: colNew && r2(colNew.dx) } : undefined,
    pages: pNew.pages,
  };
  let label;
  let reason;
  if (fx === "text-stress-ptbr") {
    label = "pdf-probe-pairing-artefact";
    reason = `Occurrence pairing across differently paginated pages: vellora lays text-stress-ptbr out on ${pNew.pages.vellora} pages, Chrome on ${pNew.pages.chrome}, so occurrence #${occ} of "${text}" sits in a row (or repeated <thead>) that Chrome puts on another page and the gate pairs it with a different word (old and new pair the same two words: vellora (${r2(nNew.vellora.x)}, ${r2(nNew.vellora.y)}) p${nNew.vellora.page} vs Chrome (${r2(nNew.chrome.x)}, ${r2(nNew.chrome.y)}) p${nNew.chrome.page}). Compared with the Chrome column the word belongs to (nearest same-text Chrome x), it moved from ${r2(colOld?.dx)} pt to ${r2(colNew?.dx)} pt: the extrato table now keeps its <col> widths (fixed_table_widths deleted).`;
  } else if (posNew != null && posOld != null && (posNew < posOld - 1e-9 || Math.abs(posNew - posOld) < 1)) {
    label = "pdf-probe-pairing-artefact";
    reason = `Occurrence pairing of an identical token: the gate pairs occurrence #${occ} of "${text}" on each side, and the vellora word order changed, so old and new compare different Chrome words (new pair: vellora (${r2(nNew.vellora.x)}, ${r2(nNew.vellora.y)}) vs Chrome (${r2(nNew.chrome.x)}, ${r2(nNew.chrome.y)})). By position (nearest same-text Chrome word on the same page) |${k}| went ${r2(posOld)} -> ${r2(posNew)} pt: ${posNew < posOld ? "closer to Chrome" : "a change below the 1 pt tolerance"}.`;
  } else if (
    k === "dy" &&
    pOld.pairs.some(
      (x) =>
        x.text === text &&
        x.vellora?.page === nNew.vellora.page &&
        Math.abs(x.vellora.y - nNew.vellora.y) < 0.5 &&
        Math.abs(Math.abs(x.vellora.y - nNew.chrome.y) - Math.abs(it.after)) < 1,
    )
  ) {
    const same = pOld.pairs.find(
      (x) => x.text === text && x.vellora?.page === nNew.vellora.page && Math.abs(x.vellora.y - nNew.vellora.y) < 0.5,
    );
    e.sameRowOld = { vellora: same.vellora, dyVsNewChromePartner: r2(same.vellora.y - nNew.chrome.y) };
    label = "pdf-probe-pairing-artefact";
    reason = `Occurrence pairing of an identical token whose old pdftotext order was scrambled: the old run listed the "${text}" of this row (vellora y ${r2(same.vellora.y)}) as another occurrence, so the old occurrence #${occ} was a word of a different row. The same row's word against the same Chrome word (y ${r2(nNew.chrome.y)}) has dy ${r2(same.vellora.y - nNew.chrome.y)} pt in the old run and ${r2(nNew.vellora.y - nNew.chrome.y)} pt now: unchanged by row (only its x moved, ${r2(same.vellora.x - nNew.chrome.x)} -> ${r2(nNew.vellora.x - nNew.chrome.x)} pt). The growing dy down the item rows is raw Blitz row height.`;
  } else if (fx === "danfe" && posNew != null && posOld != null && Math.abs(r2(posNew) - Math.abs(it.after)) < 0.01 && Math.abs(r2(posOld) - Math.abs(it.before)) < 0.01) {
    label = "coincidental-match-of-wrong-geometry";
    reason = `|${k}| ${Math.abs(it.before)} -> ${Math.abs(it.after)} pt (same word pair before and after). ${DANFE_TABLE_NOTE}`;
  } else {
    throw new Error(`unclassified ${it.key}: posOld ${posOld} posNew ${posNew}`);
  }
  e.label = label;
  evidence.push(e);
  accepted.push({
    key: it.key,
    after: it.after,
    label,
    reason,
    evidence: `pdf-probe pairs (poppler pdftotext -bbox, PDF pt) of port/02 run 20261021T102146Z and this run; Chrome 155.0.8059.85 local (INDICATIVE). Per-item numbers: attachments/accepted-evidence.json["${it.key}"]; Chrome/Blitz probe numbers: attachments/chrome-probes-table-passes.json.`,
  });
}
// The danfe "000#12 dy" item: same rows, old occurrence pairing scrambled (old 11th/12th pairs swapped).
writeFileSync(outAccept, `${JSON.stringify({ acceptedRegressions: accepted }, null, 2)}\n`);
writeFileSync(outEvidence, `${JSON.stringify(Object.fromEntries(evidence.map((e) => [e.key, e])), null, 1)}\n`);
const tally = {};
for (const a of accepted) tally[a.label] = (tally[a.label] ?? 0) + 1;
console.log(JSON.stringify(tally));
