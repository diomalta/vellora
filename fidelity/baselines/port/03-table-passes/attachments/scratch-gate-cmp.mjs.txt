// node gate-cmp.mjs <gate-report A.json> <gate-report B.json> [keyRegex]
// Both gated against the same baseline: per-key class/after differences between runs A and B.
import { readFileSync } from "node:fs";
const [a, b, re] = process.argv.slice(2);
const A = JSON.parse(readFileSync(a, "utf8"));
const B = JSON.parse(readFileSync(b, "utf8"));
const rx = re ? new RegExp(re) : null;
const ma = new Map(A.items.map((i) => [i.key, i]));
const mb = new Map(B.items.map((i) => [i.key, i]));
const rows = [];
for (const k of new Set([...ma.keys(), ...mb.keys()])) {
  if (rx && !rx.test(k)) continue;
  const x = ma.get(k);
  const y = mb.get(k);
  const ca = x ? x.class : "unchanged";
  const cb = y ? y.class : "unchanged";
  const va = x ? x.after : "(base)";
  const vb = y ? y.after : "(base)";
  if (ca === cb && JSON.stringify(va) === JSON.stringify(vb)) continue;
  rows.push({ k, ca, cb, before: (x ?? y).before, va, vb });
}
const tally = {};
for (const r of rows) {
  const t = `${r.ca} -> ${r.cb}`;
  tally[t] = (tally[t] ?? 0) + 1;
}
console.log(JSON.stringify(tally, null, 1));
for (const r of rows.filter((r) => !r.k.startsWith("pdf-probe|"))) {
  console.log(r.k, "| base", JSON.stringify(r.before), "| A", r.ca, JSON.stringify(r.va), "| B", r.cb, JSON.stringify(r.vb));
}
// pdf-probe summary per fixture: count of words where B is better/worse than A (|value| vs Chrome)
const per = {};
for (const r of rows.filter((r) => r.k.startsWith("pdf-probe|"))) {
  const fx = r.k.split("|")[1];
  const base = Math.abs(Number(r.before));
  const va = r.va === "(base)" ? base : Math.abs(Number(r.va));
  const vb = r.vb === "(base)" ? base : Math.abs(Number(r.vb));
  per[fx] ??= { bBetter: 0, bWorse: 0, same: 0 };
  if (vb < va - 1e-9) per[fx].bBetter++;
  else if (vb > va + 1e-9) per[fx].bWorse++;
  else per[fx].same++;
}
console.log("pdf-probe (B vs A, |delta| vs Chrome):", JSON.stringify(per));
